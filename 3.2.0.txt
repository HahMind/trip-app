===== ./src/App.jsx =====

import { WifiOff } from "lucide-react";
import { useCallback, useEffect, useMemo, useRef, useState } from "react";
import Swal from "sweetalert2";
import { dynaura, Toaster } from "dynaura";
import { CATEGORIES, HISTORY_CATEGORY, SWAL_THEME } from "./data/constants";
import {
  formatForDB,
  formatMoney,
  getStoredMemberDetails,
  getStoredMembers,
  getTodayISO,
  getOfflineQueue,
  parseForUI,
  saveOfflineQueue,
} from "./utils/trip";
import ExpenseForm from "./components/ExpenseForm";
import ExpenseList from "./components/ExpenseList";
import HistoryModal from "./components/HistoryModal";
import MemberModal from "./components/MemberModal";
import PaymentModal from "./components/PaymentModal";
import SplitModal from "./components/SplitModal";
import TripHeader from "./components/TripHeader";
import ThemeMenu from "./components/ThemeMenu";
import SettlementSection from "./components/SettlementSection";
import usePinchGuard from "./hooks/usePinchGuard";
import useTripData from "./hooks/useTripData";
import useSettlements from "./hooks/useSettlements";
import { supabase } from "./services/supabase";
const version_number = "3.2.1";

export default function App() {
  const [memberDetails, setMemberDetails] = useState(getStoredMemberDetails);

  const [paidBy, setPaidBy] = useState(() => {
    const saved = localStorage.getItem("defaultPaidBy");
    const storedMembers = getStoredMembers();
    return saved && storedMembers.includes(saved)
      ? saved
      : storedMembers[0] || "";
  });

  const updatePaidByFromMembers = useCallback((members) => {
    setPaidBy((previous) =>
      members.includes(previous) ? previous : members[0] || "",
    );
  }, []);
  const updateMemberDetailsFromSupabase = useCallback((details) => {
    setMemberDetails(details);
  }, []);
  const {
    members: MEMBERS,
    expenses,
    setExpenses,
    isOffline,
    loading,
    isSyncing,
    refreshMembers,
    refreshExpenses,
  } = useTripData(updatePaidByFromMembers, updateMemberDetailsFromSupabase);

  const [isListExpanded, setIsListExpanded] = useState(false);
  const [collapsedDates, setCollapsedDates] = useState({});
  const [showHistory, setShowHistory] = useState(false);
  const [showMemberModal, setShowMemberModal] = useState(false);
  const [activePayment, setActivePayment] = useState(null);
  const [activeSplitExpense, setActiveSplitExpense] = useState(null);
  const [editId, setEditId] = useState(null);
  const [swipedId, setSwipedId] = useState(null);
  const [highlightedId, setHighlightedId] = useState(null);
  const [isSubmittingExpense, setIsSubmittingExpense] = useState(false);
  const [pendingMemberAction, setPendingMemberAction] = useState(null);
  const [pendingDeleteId, setPendingDeleteId] = useState(null);
  const [pendingSplitId, setPendingSplitId] = useState(null);
  const [pendingSettlement, setPendingSettlement] = useState(null);
  const [pendingHistoryId, setPendingHistoryId] = useState(null);
  const expenseSubmitLock = useRef(false);
  const settlementSubmitLock = useRef(false);
  const syncOfflineQueueLock = useRef(false);
  const [draftSplitBetween, setDraftSplitBetween] = useState([paidBy]);
  const [didEditDraftSplit, setDidEditDraftSplit] = useState(false);
  const [didEditOdometerStart, setDidEditOdometerStart] = useState(false);
  usePinchGuard();

  const [date, setDate] = useState(getTodayISO());
  const [item, setItem] = useState("");
  const [amount, setAmount] = useState("");
  const [category, setCategory] = useState(CATEGORIES[0]);

  const [odometerStart, setOdometerStart] = useState("");
  const [odometerEnd, setOdometerEnd] = useState("");

  const handleAppClick = () => {
    if (swipedId) setSwipedId(null);
  };

  const syncOfflineQueue = useCallback(async () => {
    if (!navigator.onLine || syncOfflineQueueLock.current) return;
    const queue = getOfflineQueue();
    if (queue.length === 0) return;

    syncOfflineQueueLock.current = true;
    const sortedQueue = [...queue].sort(
      (a, b) => new Date(a.created_at) - new Date(b.created_at),
    );
    const cleanPayloads = sortedQueue.map((item) => {
      const payload = { ...item };
      delete payload.id;
      delete payload.isPendingSync;
      return payload;
    });

    try {
      const { error } = await supabase
        .from("trip_expenses")
        .insert(cleanPayloads);
      if (error) throw error;

      saveOfflineQueue([]);
      await refreshExpenses();
      dynaura.success({
        title: "ซิงค์รายการออฟไลน์สำเร็จ",
        description: `${sortedQueue.length} รายการ`,
        duration: 4000,
      });
    } catch (error) {
      dynaura.error({
        title: "ซิงค์รายการออฟไลน์ไม่สำเร็จ",
        description: error.message,
        duration: 5000,
      });
    } finally {
      syncOfflineQueueLock.current = false;
    }
  }, [refreshExpenses]);

  useEffect(() => {
    const handleOnline = () => syncOfflineQueue();
    window.addEventListener("online", handleOnline);
    if (navigator.onLine) syncOfflineQueue();
    return () => window.removeEventListener("online", handleOnline);
  }, [syncOfflineQueue]);

  useEffect(() => {
    if (editId || didEditOdometerStart) return;
    const latestGasExpense = [...expenses]
      .filter(
        (expense) =>
          expense.category === "ค่าน้ำมัน" &&
          expense.odometer_end != null &&
          expense.odometer_end !== "",
      )
      .sort(
        (a, b) =>
          new Date(b.created_at || 0) - new Date(a.created_at || 0),
      )[0];
    if (latestGasExpense) {
      setOdometerStart(String(latestGasExpense.odometer_end));
    }
  }, [expenses, editId, didEditOdometerStart]);

  useEffect(() => {
    if (!editId && !didEditDraftSplit) setDraftSplitBetween([paidBy]);
  }, [paidBy, editId, didEditDraftSplit]);

  // บันทึกตรงเข้า Database
  const handleAddMember = async (name, details) => {
    if (pendingMemberAction) return;
    // 1. ตรวจสอบสถานะการเชื่อมต่ออินเทอร์เน็ต
    if (isOffline)
      return dynaura.warning({
        title: "ออฟไลน์",
        description: "ไม่สามารถเพิ่มสมาชิกขณะออฟไลน์ได้",
        duration: 4000,
      });

    if (!MEMBERS.includes(name)) {
      setPendingMemberAction(name);
      const notificationId = dynaura.loading({
        id: `add-member-${Date.now()}`,
        title: `กำลังเพิ่ม ${name}`,
        description: "กำลังบันทึกสมาชิกลงฐานข้อมูล",
        progress: 0.15,
      });
      // 2. บันทึกรายชื่อสมาชิกใหม่ลงฐานข้อมูล Supabase
      try {
        const { error } = await supabase.from("trip_members").insert([
          {
            name,
            promptpay: details?.promptpay || null,
            bank_id: details?.bankId || null,
            bank_acc: details?.bankAcc || null,
          },
        ]);
        if (error) throw error;
        dynaura.update(notificationId, { progress: 0.72 });

        setPaidBy(name);
        localStorage.setItem("defaultPaidBy", name);

        await refreshMembers();

        if (details) handleUpdateMemberDetails(name, details, false);
        dynaura.update(notificationId, {
          type: "success",
          title: "เพิ่ม " + name + " สำเร็จ",
          description: undefined,
          progress: undefined,
          duration: 3000,
        });
      } catch (error) {
        dynaura.update(notificationId, {
          type: "error",
          title: "เพิ่มไม่สำเร็จ",
          description: error.message,
          progress: undefined,
          duration: 5000,
        });
      } finally {
        setPendingMemberAction(null);
      }
    }
  };

  const handleUpdateMemberDetails = async (
    name,
    details,
    showToast = true,
  ) => {
    try {
      const { error } = await supabase
        .from("trip_members")
        .update({
          promptpay: details.promptpay || null,
          bank_id: details.bankId || null,
          bank_acc: details.bankAcc || null,
        })
        .eq("name", name);
      if (error) throw error;

      const updated = { ...memberDetails, [name]: details };
      setMemberDetails(updated);
      localStorage.setItem("memberDetails", JSON.stringify(updated));
      if (showToast)
        dynaura.success({
          title: "บันทึกข้อมูลเรียบร้อย",
          description: `ข้อมูลการชำระเงินของ ${name}`,
          duration: 1000,
        });
    } catch (error) {
      dynaura.error({
        title: "บันทึกข้อมูลไม่สำเร็จ",
        description: error.message,
        duration: 5000,
      });
    }
  };

  const handleRemoveMember = async (name, hasDebt) => {
    if (pendingMemberAction) return;
    if (isOffline)
      return dynaura.warning({
        title: "ออฟไลน์",
        description: "ไม่สามารถลบสมาชิกขณะออฟไลน์ได้",
        duration: 4000,
      });
    if (MEMBERS.length <= 1)
      return dynaura.warning({
        title: "ลบไม่ได้",
        description: "ต้องมีสมาชิกอย่างน้อย 1 คน",
        duration: 4000,
      });
    if (hasDebt)
      return dynaura.error({
        title: "ลบไม่ได้",
        description: `${name} มียอดค้างชำระอยู่`,
        duration: 4000,
      });

    const result = await Swal.fire({
      ...SWAL_THEME,
      title: "ยืนยันการลบ",
      text: `ลบ ${name} ออกจากทริป?`,
      showCancelButton: true,
      confirmButtonText: "ลบ",
      cancelButtonText: "ยกเลิก",
      customClass: {
        ...SWAL_THEME.customClass,
        confirmButton:
          "bg-red-600 text-white font-bold px-5 py-2.5 rounded-lg hover:bg-red-500 transition text-sm cursor-pointer",
      },
    });
    if (!result.isConfirmed) return;
    setPendingMemberAction(name);
    const notificationId = dynaura.loading({
      id: `remove-member-${name}-${Date.now()}`,
      title: `กำลังลบ ${name}`,
      description: "กำลังอัปเดตรายชื่อสมาชิก",
      progress: 0.15,
    });
    try {
      const { error } = await supabase
        .from("trip_members")
        .delete()
        .eq("name", name);
      if (error) throw error;
      dynaura.update(notificationId, { progress: 0.72 });
      await refreshMembers();
      dynaura.update(notificationId, {
        type: "success",
        title: `ลบ ${name} แล้ว`,
        description: undefined,
        progress: undefined,
        duration: 3000,
      });
    } catch (error) {
      dynaura.update(notificationId, {
        type: "error",
        title: "ลบไม่สำเร็จ",
        description: error.message,
        progress: undefined,
        duration: 5000,
      });
    } finally {
      setPendingMemberAction(null);
    }
  };

  const handleAddExpense = async (event) => {
    event.preventDefault();
    if (expenseSubmitLock.current) return;

    const numericAmount = Number(amount);
    if (!Number.isFinite(numericAmount) || numericAmount <= 0)
      return Swal.fire({
        ...SWAL_THEME,
        icon: "warning",
        title: "กรุณากรอกจำนวนเงิน",
      });

    const wasEditing = Boolean(editId);
    expenseSubmitLock.current = true;
    setIsSubmittingExpense(true);
    const notificationId = dynaura.loading({
      id: `expense-${wasEditing ? "edit" : "add"}-${Date.now()}`,
      title: wasEditing ? "กำลังแก้ไขรายการ" : "กำลังบันทึกรายการ",
      description: item.trim() || category,
      progress: 0.12,
    });

    const payload = {
      expense_date: formatForDB(date),
      item_name: item.trim() || category,
      amount: numericAmount,
      category,
      paid_by: paidBy,
      split_between:
        draftSplitBetween.length > 0 ? draftSplitBetween : [paidBy],
      odometer_start:
        category === "ค่าน้ำมัน" && odometerStart
          ? Number(odometerStart)
          : null,
      odometer_end:
        category === "ค่าน้ำมัน" && odometerEnd ? Number(odometerEnd) : null,
    };

    if (isOffline && !editId) {
      const offlineExpense = {
        ...payload,
        id: `offline_${crypto.randomUUID()}`,
        isPendingSync: true,
        created_at: new Date().toISOString(),
      };
      const queue = getOfflineQueue();
      saveOfflineQueue([...queue, offlineExpense]);

      const cachedData = localStorage.getItem("tripExpensesCache");
      const cachedExpenses = cachedData ? JSON.parse(cachedData) : [];
      const baseExpenses = cachedExpenses.filter(
        (expense) => !expense.isPendingSync,
      );
      const mergedExpenses = [
        ...baseExpenses,
        ...queue,
        offlineExpense,
      ].sort(
        (a, b) => new Date(b.created_at || 0) - new Date(a.created_at || 0),
      );
      setExpenses(mergedExpenses);
      localStorage.setItem("tripExpensesCache", JSON.stringify(mergedExpenses));

      setItem("");
      setAmount("");
      setCategory(CATEGORIES[0]);
      setOdometerEnd("");
      setDraftSplitBetween([paidBy]);
      setDidEditDraftSplit(false);
      setDidEditOdometerStart(false);
      setDidEditDraftSplit(false);
      setDidEditOdometerStart(false);
      dynaura.success({
        title: "บันทึกออฟไลน์แล้ว (รอซิงค์เมื่อมีเน็ต)",
        description: `${formatMoney(numericAmount)} บาท`,
        duration: 4000,
      });
      return;
    }

    if (isOffline) {
      return Swal.fire({
        ...SWAL_THEME,
        icon: "warning",
        title: "ออฟไลน์",
        text: "ไม่สามารถแก้ไขรายการขณะออฟไลน์ได้",
      });
    }

    try {
      if (wasEditing) {
        const { error } = await supabase
          .from("trip_expenses")
          .update(payload)
          .eq("id", editId);
        if (error) throw error;
        dynaura.update(notificationId, { progress: 0.72 });
        setEditId(null);
        setHighlightedId(editId);
        setTimeout(() => setHighlightedId(null), 2000);
      } else {
        const { error } = await supabase.from("trip_expenses").insert([payload]);
        if (error) throw error;
        dynaura.update(notificationId, { progress: 0.72 });
      }

      await refreshExpenses();
      dynaura.update(notificationId, { progress: 1 });
      setItem("");
      setAmount("");
      setCategory(CATEGORIES[0]);
      setOdometerEnd("");
      setDraftSplitBetween([paidBy]);
      dynaura.update(notificationId, {
        type: "success",
        title: wasEditing ? "แก้ไขสำเร็จ" : "บันทึกสำเร็จ",
        description: `${formatMoney(numericAmount)} บาท`,
        progress: undefined,
        duration: 4000,
        confetti: true,
      });
    } catch (error) {
      dynaura.update(notificationId, {
        type: "error",
        title: wasEditing ? "แก้ไขไม่สำเร็จ" : "บันทึกไม่สำเร็จ",
        description: error.message,
        progress: undefined,
        duration: 5000,
      });
    } finally {
      expenseSubmitLock.current = false;
      setIsSubmittingExpense(false);
    }
  };

  const handleUpdateSplit = async (id, newSplit) => {
    if (id === null) {
      setDraftSplitBetween(newSplit);
      setDidEditDraftSplit(true);
      setActiveSplitExpense(null);
      return true;
    }
    if (pendingSplitId) return false;
    if (isOffline) {
      dynaura.warning({
        title: "ออฟไลน์",
        description: "ไม่สามารถแก้ไขการหารขณะออฟไลน์ได้",
        duration: 4000,
      });
      return false;
    }
    setPendingSplitId(id);
    const notificationId = dynaura.loading({
      id: `update-split-${id}-${Date.now()}`,
      title: "กำลังอัปเดตการหาร",
      description: "กำลังบันทึกผู้ร่วมรายการ",
      progress: 0.15,
    });
    try {
      const { error } = await supabase
        .from("trip_expenses")
        .update({ split_between: newSplit })
        .eq("id", id);
      if (error) throw error;
      dynaura.update(notificationId, { progress: 0.72 });
      await refreshExpenses();
      dynaura.update(notificationId, {
        type: "success",
        title: "อัปเดตการหารสำเร็จ",
        description: "ยอดแบ่งรายการได้รับการอัปเดต",
        progress: undefined,
        duration: 3000,
      });
      return true;
    } catch (error) {
      dynaura.update(notificationId, {
        type: "error",
        title: "อัปเดตไม่สำเร็จ",
        description: error.message,
        progress: undefined,
        duration: 5000,
      });
      return false;
    } finally {
      setPendingSplitId(null);
    }
  };

  const cancelEdit = () => {
    setEditId(null);
    setDate(getTodayISO());
    setItem("");
    setAmount("");
    setCategory(CATEGORIES[0]);
    setOdometerStart("");
    setOdometerEnd("");
    setDidEditOdometerStart(false);
    setDidEditDraftSplit(false);
  };

  const handleDelete = async (id) => {
    if (pendingDeleteId) return;
    if (isOffline)
      return dynaura.warning({
        title: "ออฟไลน์",
        description: "ไม่สามารถลบรายการขณะออฟไลน์ได้",
        duration: 4000,
      });
    const r = await Swal.fire({
      ...SWAL_THEME,
      title: "ยืนยันการลบ?",
      showCancelButton: true,
      confirmButtonText: "ลบ",
      customClass: {
        ...SWAL_THEME.customClass,
        confirmButton:
          "bg-red-600 text-white font-bold px-5 py-2.5 rounded-lg hover:bg-red-500 transition text-sm cursor-pointer",
      },
    });
    if (r.isConfirmed) {
      setPendingDeleteId(id);
      const notificationId = dynaura.loading({
        id: `delete-expense-${id}-${Date.now()}`,
        title: "กำลังลบรายการ",
        description: "กำลังอัปเดตรายจ่ายและยอดคงเหลือ",
        progress: 0.15,
      });
      try {
        const { error } = await supabase
          .from("trip_expenses")
          .delete()
          .eq("id", id);
        if (error) throw error;
        dynaura.update(notificationId, { progress: 0.72 });
        await refreshExpenses();
        dynaura.update(notificationId, {
          type: "success",
          title: "ลบรายการแล้ว",
          description: "รายจ่ายและยอดคงเหลือถูกอัปเดต",
          progress: undefined,

          progress: undefined,
          duration: 5000,
        });

      }

      )}
    </div>
  );
}

export default function ExpenseItem({

      duration: 4000,

  //  สั่ง Auto-scroll มาที่ Item นี้เมื่อโดน Highlight

      });
    }
  }, [isHighlighted]);

  const splitList = validSplit.length > 0

import { X } from "lucide-react";

import { formatMoney } from "../utils/trip";
import ModalShell from "./ModalShell";

export default function SplitModal({ expense, members, onClose, onSave }) {
  const [isSaving, setIsSaving] = useState(false);
  const [selectedSplit, setSelectedSplit] = useState(() => {
    const rawSplit =
      expense?.split_between?.length > 0
        ? expense.split_between
        : [expense.paid_by];
    return rawSplit.filter((m) => members.includes(m));
  });

  useEffect(() => {
    const handleEsc = (e) => {
      if (e.key === "Escape") onClose();
    };
    document.addEventListener("keydown", handleEsc);
    return () => document.removeEventListener("keydown", handleEsc);
  }, [onClose]);

  const handleToggle = (member) =>
    setSelectedSplit((prev) =>
      prev.includes(member)
        ? prev.filter((m) => m !== member)
        : [...prev, member],
    );
  const isAllSelected =
    selectedSplit.length === members.length && members.length > 0;
  const handleToggleAll = () =>
    isAllSelected ? setSelectedSplit([]) : setSelectedSplit([...members]);

import { createContext, useContext, useEffect, useState } from "react";

const ThemeContext = createContext(null);

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState(
    () => localStorage.getItem("tripTheme") || "pink",
  );

  useEffect(() => {
    document.documentElement.dataset.theme = theme;
    localStorage.setItem("tripTheme", theme);

    const themeColors = {
      dark: "#111827",
      pink: "#fec7d7",
      light: "#55423d",
      purple: "#faeee7",
    };
    const themeColor = themeColors[theme] || themeColors.dark;

    let themeMeta = document.querySelector('meta[name="theme-color"]');
    if (!themeMeta) {
      themeMeta = document.createElement("meta");
      themeMeta.name = "theme-color";
      document.head.appendChild(themeMeta);
    }
    themeMeta.content = themeColor;

    const statusBarMeta = document.querySelector(
      'meta[name="apple-mobile-web-app-status-bar-style"]',
    );
    if (statusBarMeta) {
      statusBarMeta.content = theme === "dark" ? "black" : "default";
    }
  }, [theme]);

  return (
    <ThemeContext.Provider value={{ theme, setTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme() {
  const context = useContext(ThemeContext);
  if (!context) throw new Error("useTheme must be used within ThemeProvider");
  return context;
}

import { useEffect } from "react";

export default function usePinchGuard() {
  useEffect(() => {
    const preventPinchTouch = (event) => {
      if (event.touches?.length > 1) event.preventDefault();
    };
    const preventSafariGesture = (event) => event.preventDefault();

    document.addEventListener("touchmove", preventPinchTouch, { passive: false });
    document.addEventListener("gesturestart", preventSafariGesture, {
      passive: false,
    });
    document.addEventListener("gesturechange", preventSafariGesture, {
      passive: false,
    });

    return () => {
      document.removeEventListener("touchmove", preventPinchTouch);
      document.removeEventListener("gesturestart", preventSafariGesture);
      document.removeEventListener("gesturechange", preventSafariGesture);
    };
  }, []);
}


===== ./src/hooks/useSettlements.js =====

import { useMemo } from "react";
import { HISTORY_CATEGORY } from "../data/constants";

export default function useSettlements(expenses, members) {
  return useMemo(() => {
    const regularExpenses = expenses.filter(
      (expense) => expense.category !== HISTORY_CATEGORY,
    );
    const historyExpenses = expenses.filter(
      (expense) => expense.category === HISTORY_CATEGORY,
    );

    const balances = members.map((member) => ({ member, balance: 0 }));

    regularExpenses.forEach((expense) => {
      const payer = balances.find((balance) => balance.member === expense.paid_by);
      if (payer) payer.balance += Number(expense.amount);

      const configuredSplit = Array.isArray(expense.split_between)
        ? expense.split_between
        : [];
      const validSplit = configuredSplit.filter((member) =>
        members.includes(member),
      );
      const splitters =
        validSplit.length > 0
          ? validSplit
          : members.includes(expense.paid_by)
            ? [expense.paid_by]
            : [members[0]];

      const costPerPerson = Number(expense.amount) / splitters.length;
      splitters.forEach((member) => {
        const debtor = balances.find((balance) => balance.member === member);
        if (debtor) debtor.balance -= costPerPerson;
      });
    });

    historyExpenses.forEach((expense) => {
      const payer = balances.find((balance) => balance.member === expense.paid_by);
      if (payer) payer.balance += Number(expense.amount);
      members.forEach((member) => {
        if (expense.item_name.includes(`-> ${member}`)) {
          const receiver = balances.find(
            (balance) => balance.member === member,
          );
          if (receiver) receiver.balance -= Number(expense.amount);
        }
      });
    });

    const debtors = balances
      .filter((balance) => balance.balance < -0.01)
      .map((balance) => ({ ...balance }));
    const creditors = balances
      .filter((balance) => balance.balance > 0.01)
      .map((balance) => ({ ...balance }));
    const transfers = [];
    let debtorIndex = 0;
    let creditorIndex = 0;

    while (debtorIndex < debtors.length && creditorIndex < creditors.length) {
      const debtor = debtors[debtorIndex];
      const creditor = creditors[creditorIndex];
      const amount = Math.min(Math.abs(debtor.balance), creditor.balance);
      if (amount > 0.01) {
        transfers.push({ from: debtor.member, to: creditor.member, amount });
      }
      debtor.balance += amount;
      creditor.balance -= amount;
      if (Math.abs(debtor.balance) < 0.01) debtorIndex++;
      if (creditor.balance < 0.01) creditorIndex++;
    }

    return { balances, transfers };
  }, [expenses, members]);
}


===== ./src/hooks/useSwipeDrag.js =====

import { useRef, useState } from "react";

export default function useSwipeDrag({
  id,
  isOpen,
  setOpen,
  canDrag = true,
}) {
  const [drag, setDrag] = useState({
    active: false,
    startX: 0,
    startY: 0,
    offset: 0,
    scrolling: false,
  });
  const isMouse = useRef(false);
  const suppressClick = useRef(false);

  let currentOffset = isOpen ? -70 : 0;
  if (drag.active) {
    currentOffset = (isOpen ? -70 : 0) + drag.offset;
    currentOffset = Math.max(-90, Math.min(0, currentOffset));
  }

  const startDrag = (clientX, clientY) => {
    if (!canDrag) return;
    setDrag({
      active: true,
      startX: clientX,
      startY: clientY,
      offset: 0,
      scrolling: false,
    });
    if (!isOpen) setOpen(null);
  };

  const moveDrag = (clientX, clientY) => {
    if (!drag.active) return;
    const diffX = clientX - drag.startX;
    const diffY = Math.abs(clientY - drag.startY);

    if (!drag.scrolling && diffY > 15 && Math.abs(diffX) < 15) {
      setDrag((previous) => ({ ...previous, scrolling: true }));
      return;
    }
    if (!drag.scrolling) {
      setDrag((previous) => ({ ...previous, offset: diffX }));
    }
  };

  const endDrag = () => {
    if (!drag.active) return;
    suppressClick.current = Math.abs(drag.offset) > 5;
    isMouse.current = false;

    if (!drag.scrolling) {
      const baseOffset = isOpen ? -70 : 0;
      if (baseOffset + drag.offset < -35) setOpen(id);
      else setOpen(null);
    }

    setDrag({
      active: false,
      startX: 0,
      startY: 0,
      offset: 0,
      scrolling: false,
    });
  };

  const touchHandlers = {
    onTouchStart: (event) =>
      startDrag(
        event.targetTouches[0].clientX,
        event.targetTouches[0].clientY,
      ),
    onTouchMove: (event) =>
      moveDrag(
        event.targetTouches[0].clientX,
        event.targetTouches[0].clientY,
      ),
    onTouchEnd: endDrag,
  };

  const mouseHandlers = {
    onMouseDown: (event) => {
      isMouse.current = true;
      startDrag(event.clientX, event.clientY);
    },
    onMouseMove: (event) => {
      if (isMouse.current) moveDrag(event.clientX, event.clientY);
    },
    onMouseUp: () => {
      if (isMouse.current) endDrag();
    },
    onMouseLeave: () => {
      if (isMouse.current) endDrag();
    },
  };

  const consumeSuppressedClick = () => {
    if (!suppressClick.current) return false;
    suppressClick.current = false;
    return true;
  };

  return {
    drag,
    currentOffset,
    isSwiped: currentOffset < -5,
    consumeSuppressedClick,
    touchHandlers,
    mouseHandlers,
  };
}


===== ./src/hooks/useTripData.js =====

import { useCallback, useEffect, useState } from "react";
import { getOfflineQueue, getStoredMembers } from "../utils/trip";
import { supabase } from "../services/supabase";

export default function useTripData(onMembersUpdated, onMemberDetailsUpdated) {
  const [members, setMembers] = useState(getStoredMembers);
  const [expenses, setExpenses] = useState([]);
  const [isOffline, setIsOffline] = useState(!navigator.onLine);
  const [loading, setLoading] = useState(true);
  const [isSyncing, setIsSyncing] = useState(false);

  const refreshMembers = useCallback(async () => {
    if (!navigator.onLine) return;

    const { data } = await supabase
      .from("trip_members")
      .select("name, promptpay, bank_id, bank_acc")
      .order("created_at", { ascending: true });

    if (data && data.length > 0) {
      const fetchedMembers = data.map((member) => member.name);
      const fetchedDetails = data.reduce((details, member) => {
        details[member.name] = {
          promptpay: member.promptpay || "",
          bankId: member.bank_id || "kbank",
          bankAcc: member.bank_acc || "",
        };
        return details;
      }, {});
      setMembers(fetchedMembers);
      localStorage.setItem("tripMembers", JSON.stringify(fetchedMembers));
      localStorage.setItem("memberDetails", JSON.stringify(fetchedDetails));
      onMembersUpdated(fetchedMembers);
      if (onMemberDetailsUpdated) onMemberDetailsUpdated(fetchedDetails);
    }
  }, [onMembersUpdated, onMemberDetailsUpdated]);

  const refreshExpenses = useCallback(async () => {
    if (!navigator.onLine) {
      const cachedData = localStorage.getItem("tripExpensesCache");
      const cachedExpenses = cachedData ? JSON.parse(cachedData) : [];
      const queuedExpenses = getOfflineQueue();
      const mergedById = new Map();
      [...cachedExpenses, ...queuedExpenses].forEach((expense) => {
        mergedById.set(expense.id, expense);
      });
      const mergedExpenses = [...mergedById.values()].sort(
        (a, b) => new Date(b.created_at || 0) - new Date(a.created_at || 0),
      );
      setExpenses(mergedExpenses);
      localStorage.setItem("tripExpensesCache", JSON.stringify(mergedExpenses));
      return;
    }

    setIsSyncing(true);
    try {
      const { data, error } = await supabase
        .from("trip_expenses")
        .select("*")
        .order("created_at", { ascending: false });

      if (error) throw error;

      if (data) {
        setExpenses(data);
        localStorage.setItem("tripExpensesCache", JSON.stringify(data));
      }
    } catch {
      const cachedData = localStorage.getItem("tripExpensesCache");
      if (cachedData) setExpenses(JSON.parse(cachedData));
    } finally {
      setTimeout(() => setIsSyncing(false), 500);
    }
  }, []);

  useEffect(() => {
    setIsOffline(!navigator.onLine);

    const handleOnline = () => {
      setIsOffline(false);
      Promise.all([refreshMembers(), refreshExpenses()]).then(() =>
        setLoading(false),
      );
    };
    const handleOffline = () => setIsOffline(true);

    window.addEventListener("online", handleOnline);
    window.addEventListener("offline", handleOffline);

    Promise.all([refreshMembers(), refreshExpenses()]).then(() =>
      setLoading(false),
    );

    const expenseSub = supabase
      .channel("trip_expenses_changes")
      .on(
        "postgres_changes",
        { event: "*", schema: "public", table: "trip_expenses" },
        refreshExpenses,
      )
      .subscribe();
    const memberSub = supabase
      .channel("trip_members_changes")
      .on(
        "postgres_changes",
        { event: "*", schema: "public", table: "trip_members" },
        refreshMembers,
      )
      .subscribe();

    return () => {
      window.removeEventListener("online", handleOnline);
      window.removeEventListener("offline", handleOffline);
      supabase.removeChannel(expenseSub);
      supabase.removeChannel(memberSub);
    };
  }, [refreshExpenses, refreshMembers]);

  return {
    members,
    setMembers,
    expenses,
    setExpenses,
    isOffline,
    loading,
    isSyncing,
    refreshMembers,
    refreshExpenses,
  };
}

import { dynaura } from "dynaura";
import { DEFAULT_MEMBERS } from "../data/constants";

export const getTodayISO = () => new Date().toLocaleDateString("en-CA");

export const formatForDB = (isoDate) => {
  if (!isoDate || !isoDate.includes("-")) return isoDate;
  const [y, m, d] = isoDate.split("-");
  return `${d}/${m}/${y}`;
};

export const parseForUI = (thDate) => {
  if (!thDate || !thDate.includes("/")) return getTodayISO();
  const [d, m, y] = thDate.split("/");
  return `${y}-${m.padStart(2, "0")}-${d.padStart(2, "0")}`;
};

export const getStoredMembers = () => {
  try {
    const saved = JSON.parse(localStorage.getItem("tripMembers") || "null");
    if (Array.isArray(saved) && saved.length > 0) return saved;
  } catch {
    // Ignore malformed local data and use the defaults.
  }
  return DEFAULT_MEMBERS;
};

export const getStoredMemberDetails = () => {
  try {
    const saved = JSON.parse(localStorage.getItem("memberDetails") || "null");
    if (saved) return saved;
  } catch {
    // Ignore malformed local data and use an empty details object.
  }
  return {};
};

export const formatMoney = (value) =>
  Number(value).toLocaleString(undefined, {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  });

export const normalizePromptPay = (value) => {
  let normalized = String(value || "").trim().replace(/[\s-]/g, "");
  if (normalized.startsWith("+66")) {
    normalized = `0${normalized.slice(3)}`;
  } else if (normalized.startsWith("66") && normalized.length === 11) {
    normalized = `0${normalized.slice(2)}`;
  }

  return normalized;
};

export const validatePromptPay = (value) => {
  const raw = String(value || "").trim();
  if (raw && /[^0-9+\-\s]/.test(raw)) {
    return {
      valid: false,
      normalized: "",
      message: "กรุณากรอกเฉพาะตัวเลข เว้นวรรคหรือเครื่องหมาย - เท่านั้น",
    };
  }
  if (raw.includes("+") && !raw.replace(/[\s-]/g, "").startsWith("+66")) {
    return {
      valid: false,
      normalized: "",
      message: "กรุณากรอกเบอร์โทรหรือเลขบัตรประชาชนที่ถูกต้อง",
    };
  }

  const normalized = normalizePromptPay(value);
  if (!normalized) return { valid: true, normalized: "" };

  const valid =
    (normalized.length === 10 && /^0\d{9}$/.test(normalized)) ||
    (normalized.length === 13 && /^\d{13}$/.test(normalized));

  return {
    valid,
    normalized,
    message: "กรุณากรอกพร้อมเพย์เป็นเบอร์มือถือ 10 หลักหรือเลขบัตรประชาชน 13 หลัก",
  };
};

export const openBankApp = async (textToCopy, scheme, title) => {
  let copied = false;
  if (navigator.clipboard?.writeText) {
    try {
      await navigator.clipboard.writeText(textToCopy);
      copied = true;
    } catch {
      copied = false;
    }
  } else {
    const textarea = document.createElement("textarea");
    textarea.value = textToCopy;
    textarea.style.position = "fixed";
    textarea.style.opacity = "0";
    document.body.appendChild(textarea);
    textarea.select();
    copied = document.execCommand("copy");
    textarea.remove();
  }

  if (!copied) {
    dynaura.error({
      title: "คัดลอกไม่สำเร็จ",
      description: "กรุณาลองคัดลอกเลขบัญชีอีกครั้ง",
      duration: 4000,
    });
    return;
  }

  dynaura.success({
    title,
    description: scheme
      ? "ก๊อปเลขบัญชีแล้ว กำลังเปิดแอปธนาคาร"
      : "คัดลอกเลขบัญชีเรียบร้อยแล้ว",
    duration: 3000,
  });

  if (scheme) window.location.href = scheme;
};
