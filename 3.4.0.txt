===== src/App.jsx =====
import { WifiOff } from "lucide-react";
import { useCallback, useEffect, useMemo, useRef, useState } from "react";
import Swal from "sweetalert2";
import { dynaura, Toaster } from "dynaura";
import { CATEGORIES, HISTORY_CATEGORY, SWAL_THEME } from "./data/constants";
import {
  formatForDB,
  formatMoney,
  getStoredMemberDetails,
  getStoredMembers,
  getTodayISO,
  getOfflineQueue,
  parseForUI,
  saveOfflineQueue,
} from "./utils/trip";
import SettlementDetailsModal from "./components/SettlementDetailsModal";
import ExpenseForm from "./components/ExpenseForm";
import ExpenseList from "./components/ExpenseList";
import HistoryModal from "./components/HistoryModal";
import MemberModal from "./components/MemberModal";
import PaymentModal from "./components/PaymentModal";
import SplitModal from "./components/SplitModal";
import TripHeader from "./components/TripHeader";
import ThemeMenu from "./components/ThemeMenu";
import SettlementSection from "./components/SettlementSection";
import usePinchGuard from "./hooks/usePinchGuard";
import useTripData from "./hooks/useTripData";
import useSettlements from "./hooks/useSettlements";
import { supabase } from "./services/supabase";
const version_number = "3.4.0";
export default function App() {
  const [memberDetails, setMemberDetails] = useState(getStoredMemberDetails);

  const [paidBy, setPaidBy] = useState(() => {
    const saved = localStorage.getItem("defaultPaidBy");
    const storedMembers = getStoredMembers();
    return saved && storedMembers.includes(saved)
      ? saved
      : storedMembers[0] || "";
  });

  const updatePaidByFromMembers = useCallback((members) => {
    setPaidBy((previous) =>
      members.includes(previous) ? previous : members[0] || "",
    );
  }, []);
  const updateMemberDetailsFromSupabase = useCallback((details) => {
    setMemberDetails(details);
  }, []);
  const {
    members: MEMBERS,
    expenses,
    setExpenses,
    isOffline,
    loading,
    isSyncing,
    refreshMembers,
    refreshExpenses,
  } = useTripData(updatePaidByFromMembers, updateMemberDetailsFromSupabase);

  const [isListExpanded, setIsListExpanded] = useState(false);
  const [collapsedDates, setCollapsedDates] = useState({});
  const [showHistory, setShowHistory] = useState(false);
  const [showMemberModal, setShowMemberModal] = useState(false);
  const [activePayment, setActivePayment] = useState(null);
  const [activeSettlementDetails, setActiveSettlementDetails] = useState(null);
  const [activeSplitExpense, setActiveSplitExpense] = useState(null);
  const [editId, setEditId] = useState(null);
  const [swipedId, setSwipedId] = useState(null);
  const [highlightedId, setHighlightedId] = useState(null);
  const [isSubmittingExpense, setIsSubmittingExpense] = useState(false);
  const [pendingMemberAction, setPendingMemberAction] = useState(null);
  const [pendingDeleteId, setPendingDeleteId] = useState(null);
  const [pendingSplitId, setPendingSplitId] = useState(null);
  const [pendingSettlement, setPendingSettlement] = useState(null);
  const [pendingHistoryId, setPendingHistoryId] = useState(null);
  const expenseSubmitLock = useRef(false);
  const settlementSubmitLock = useRef(false);
  const syncOfflineQueueLock = useRef(false);
  const [draftSplitBetween, setDraftSplitBetween] = useState([paidBy]);
  const [didEditDraftSplit, setDidEditDraftSplit] = useState(false);
  const [didEditOdometerStart, setDidEditOdometerStart] = useState(false);
  usePinchGuard();

  const [date, setDate] = useState(getTodayISO());
  const [item, setItem] = useState("");
  const [amount, setAmount] = useState("");
  const [category, setCategory] = useState(CATEGORIES[0]);

  const [odometerStart, setOdometerStart] = useState("");
  const [odometerEnd, setOdometerEnd] = useState("");

  const handleAppClick = () => {
    if (swipedId) setSwipedId(null);
  };

  const syncOfflineQueue = useCallback(async () => {
    if (!navigator.onLine || syncOfflineQueueLock.current) return;
    const queue = getOfflineQueue();
    if (queue.length === 0) return;

    syncOfflineQueueLock.current = true;
    const sortedQueue = [...queue].sort(
      (a, b) => new Date(a.created_at) - new Date(b.created_at),
    );
    const cleanPayloads = sortedQueue.map((item) => {
      const payload = { ...item };
      delete payload.id;
      delete payload.isPendingSync;
      return payload;
    });

    try {
      const { error } = await supabase
        .from("trip_expenses")
        .insert(cleanPayloads);
      if (error) throw error;

      saveOfflineQueue([]);
      await refreshExpenses();
      dynaura.success({
        title: "ซิงค์รายการออฟไลน์สำเร็จ",
        description: `${sortedQueue.length} รายการ`,
        duration: 4000,
      });
    } catch (error) {
      dynaura.error({
        title: "ซิงค์รายการออฟไลน์ไม่สำเร็จ",
        description: error.message,
        duration: 5000,
      });
    } finally {
      syncOfflineQueueLock.current = false;
    }
  }, [refreshExpenses]);

  useEffect(() => {
    const handleOnline = () => syncOfflineQueue();
    window.addEventListener("online", handleOnline);
    if (navigator.onLine) syncOfflineQueue();
    return () => window.removeEventListener("online", handleOnline);
  }, [syncOfflineQueue]);

  useEffect(() => {
    if (editId || didEditOdometerStart) return;
    const latestGasExpense = [...expenses]
      .filter(
        (expense) =>
          expense.category === "ค่าน้ำมัน" &&
          expense.odometer_end != null &&
          expense.odometer_end !== "",
      )
      .sort(
        (a, b) => new Date(b.created_at || 0) - new Date(a.created_at || 0),
      )[0];
    if (latestGasExpense) {
      setOdometerStart(String(latestGasExpense.odometer_end));
    }
  }, [expenses, editId, didEditOdometerStart]);

  useEffect(() => {
    if (!editId && !didEditDraftSplit) setDraftSplitBetween([paidBy]);
  }, [paidBy, editId, didEditDraftSplit]);

  // บันทึกตรงเข้า Database
  const handleAddMember = async (name, details) => {
    if (pendingMemberAction) return;
    // 1. ตรวจสอบสถานะการเชื่อมต่ออินเทอร์เน็ต
    if (isOffline)
      return dynaura.warning({
        title: "ออฟไลน์",
        description: "ไม่สามารถเพิ่มสมาชิกขณะออฟไลน์ได้",
        duration: 4000,
      });

    if (!MEMBERS.includes(name)) {
      setPendingMemberAction(name);
      const notificationId = dynaura.loading({
        id: `add-member-${Date.now()}`,
        title: `กำลังเพิ่ม ${name}`,
        description: "กำลังบันทึกสมาชิกลงฐานข้อมูล",
        progress: 0.15,
      });
      // 2. บันทึกรายชื่อสมาชิกใหม่ลงฐานข้อมูล Supabase
      try {
        const { error } = await supabase.from("trip_members").insert([
          {
            name,
            promptpay: details?.promptpay || null,
            bank_id: details?.bankId || null,
            bank_acc: details?.bankAcc || null,
          },
        ]);
        if (error) throw error;
        dynaura.update(notificationId, { progress: 0.72 });

        setPaidBy(name);
        localStorage.setItem("defaultPaidBy", name);

        await refreshMembers();

        if (details) handleUpdateMemberDetails(name, details, false);
        dynaura.update(notificationId, {
          type: "success",
          title: "เพิ่ม " + name + " สำเร็จ",
          description: undefined,
          progress: undefined,
          duration: 3000,
        });
      } catch (error) {
        dynaura.update(notificationId, {
          type: "error",
          title: "เพิ่มไม่สำเร็จ",
          description: error.message,
          progress: undefined,
          duration: 5000,
        });
      } finally {
        setPendingMemberAction(null);
      }
    }
  };

  const handleUpdateMemberDetails = async (name, details, showToast = true) => {
    try {
      const { error } = await supabase
        .from("trip_members")
        .update({
          promptpay: details.promptpay || null,
          bank_id: details.bankId || null,
          bank_acc: details.bankAcc || null,
        })
        .eq("name", name);
      if (error) throw error;

      const updated = { ...memberDetails, [name]: details };
      setMemberDetails(updated);
      localStorage.setItem("memberDetails", JSON.stringify(updated));
      if (showToast)
        dynaura.success({
          title: "บันทึกข้อมูลเรียบร้อย",
          description: `ข้อมูลการชำระเงินของ ${name}`,
          duration: 1000,
        });
    } catch (error) {
      dynaura.error({
        title: "บันทึกข้อมูลไม่สำเร็จ",
        description: error.message,
        duration: 5000,
      });
    }
  };

  const handleRemoveMember = async (name, hasDebt) => {
    if (pendingMemberAction) return;
    if (isOffline)
      return dynaura.warning({
        title: "ออฟไลน์",
        description: "ไม่สามารถลบสมาชิกขณะออฟไลน์ได้",
        duration: 4000,
      });
    if (MEMBERS.length <= 1)
      return dynaura.warning({
        title: "ลบไม่ได้",
        description: "ต้องมีสมาชิกอย่างน้อย 1 คน",
        duration: 4000,
      });
    if (hasDebt)
      return dynaura.error({
        title: "ลบไม่ได้",
        description: `${name} มียอดค้างชำระอยู่`,
        duration: 4000,
      });

    const result = await Swal.fire({
      ...SWAL_THEME,
      title: "ยืนยันการลบ",
      text: `ลบ ${name} ออกจากทริป?`,
      showCancelButton: true,
      confirmButtonText: "ลบ",
      cancelButtonText: "ยกเลิก",
      customClass: {
        ...SWAL_THEME.customClass,
        confirmButton:
          "bg-red-600 text-white font-bold px-5 py-2.5 rounded-lg hover:bg-red-500 transition text-sm cursor-pointer",
      },
    });
    if (!result.isConfirmed) return;
    setPendingMemberAction(name);
    const notificationId = dynaura.loading({
      id: `remove-member-${name}-${Date.now()}`,
      title: `กำลังลบ ${name}`,
      description: "กำลังอัปเดตรายชื่อสมาชิก",
      progress: 0.15,
    });
    try {
      const { error } = await supabase
        .from("trip_members")
        .delete()
        .eq("name", name);
      if (error) throw error;
      dynaura.update(notificationId, { progress: 0.72 });
      await refreshMembers();
      dynaura.update(notificationId, {
        type: "success",
        title: `ลบ ${name} แล้ว`,
        description: undefined,
        progress: undefined,
        duration: 3000,
      });
    } catch (error) {
      dynaura.update(notificationId, {
        type: "error",
        title: "ลบไม่สำเร็จ",
        description: error.message,
        progress: undefined,
        duration: 5000,
      });
    } finally {
      setPendingMemberAction(null);
    }
  };

  const handleAddExpense = async (event) => {
    event.preventDefault();
    if (expenseSubmitLock.current) return;

    const numericAmount = Number(amount);
    if (!Number.isFinite(numericAmount) || numericAmount <= 0)
      return Swal.fire({
        ...SWAL_THEME,
        icon: "warning",
        title: "กรุณากรอกจำนวนเงิน",
      });

    const wasEditing = Boolean(editId);
    const payload = {
      expense_date: formatForDB(date),
      item_name: item.trim() || category,
      amount: numericAmount,
      category,
      paid_by: paidBy,
      split_between:
        draftSplitBetween.length > 0 ? draftSplitBetween : [paidBy],
      odometer_start:
        category === "ค่าน้ำมัน" && odometerStart
          ? Number(odometerStart)
          : null,
      odometer_end:
        category === "ค่าน้ำมัน" && odometerEnd ? Number(odometerEnd) : null,
    };

    if (isOffline && !editId) {
      try {
        expenseSubmitLock.current = true;
        setIsSubmittingExpense(true);

        const offlineExpense = {
          ...payload,
          id: `offline_${crypto.randomUUID()}`,
          isPendingSync: true,
          created_at: new Date().toISOString(),
        };
        const queue = getOfflineQueue();
        saveOfflineQueue([...queue, offlineExpense]);

        const cachedData = localStorage.getItem("tripExpensesCache");
        const cachedExpenses = cachedData ? JSON.parse(cachedData) : [];
        const baseExpenses = cachedExpenses.filter(
          (expense) => !expense.isPendingSync,
        );
        const mergedExpenses = [...baseExpenses, ...queue, offlineExpense].sort(
          (a, b) => new Date(b.created_at || 0) - new Date(a.created_at || 0),
        );
        setExpenses(mergedExpenses);
        localStorage.setItem(
          "tripExpensesCache",
          JSON.stringify(mergedExpenses),
        );

        setItem("");
        setAmount("");
        setCategory(CATEGORIES[0]);
        setOdometerEnd("");
        setDraftSplitBetween([paidBy]);
        setDidEditDraftSplit(false);
        setDidEditOdometerStart(false);
        dynaura.success({
          title: "บันทึกออฟไลน์แล้ว (รอซิงค์เมื่อมีเน็ต)",
          description: `${formatMoney(numericAmount)} บาท`,
          duration: 4000,
        });
      } finally {
        expenseSubmitLock.current = false;
        setIsSubmittingExpense(false);
      }
      return;
    }

    if (isOffline) {
      return Swal.fire({
        ...SWAL_THEME,
        icon: "warning",
        title: "ออฟไลน์",
        text: "ไม่สามารถแก้ไขรายการขณะออฟไลน์ได้",
      });
    }

    expenseSubmitLock.current = true;
    setIsSubmittingExpense(true);
    const notificationId = dynaura.loading({
      id: `expense-${wasEditing ? "edit" : "add"}-${Date.now()}`,
      title: wasEditing ? "กำลังแก้ไขรายการ" : "กำลังบันทึกรายการ",
      description: item.trim() || category,
      progress: 0.12,
    });

    try {
      if (wasEditing) {
        const { error } = await supabase
          .from("trip_expenses")
          .update(payload)
          .eq("id", editId);
        if (error) throw error;
        dynaura.update(notificationId, { progress: 0.72 });
        setEditId(null);
        setHighlightedId(editId);
        setTimeout(() => setHighlightedId(null), 2000);
      } else {
        const { error } = await supabase
          .from("trip_expenses")
          .insert([payload]);
        if (error) throw error;
        dynaura.update(notificationId, { progress: 0.72 });
      }

      await refreshExpenses();
      dynaura.update(notificationId, { progress: 1 });
      setItem("");
      setAmount("");
      setCategory(CATEGORIES[0]);
      setOdometerEnd("");
      setDraftSplitBetween([paidBy]);
      dynaura.update(notificationId, {
        type: "success",
        title: wasEditing ? "แก้ไขสำเร็จ" : "บันทึกสำเร็จ",
        description: `${formatMoney(numericAmount)} บาท`,
        progress: undefined,
        duration: 4000,
        confetti: true,
      });
    } catch (error) {
      dynaura.update(notificationId, {
        type: "error",
        title: wasEditing ? "แก้ไขไม่สำเร็จ" : "บันทึกไม่สำเร็จ",
        description: error.message,
        progress: undefined,
        duration: 5000,
      });
    } finally {
      expenseSubmitLock.current = false;
      setIsSubmittingExpense(false);
    }
  };

  const handleUpdateSplit = async (id, newSplit) => {
    if (id === null) {
      setDraftSplitBetween(newSplit);
      setDidEditDraftSplit(true);
      setActiveSplitExpense(null);
      return true;
    }
    if (pendingSplitId) return false;
    if (isOffline) {
      dynaura.warning({
        title: "ออฟไลน์",
        description: "ไม่สามารถแก้ไขการหารขณะออฟไลน์ได้",
        duration: 4000,
      });
      return false;
    }
    setPendingSplitId(id);
    const notificationId = dynaura.loading({
      id: `update-split-${id}-${Date.now()}`,
      title: "กำลังอัปเดตการหาร",
      description: "กำลังบันทึกผู้ร่วมรายการ",
      progress: 0.15,
    });
    try {
      const { error } = await supabase
        .from("trip_expenses")
        .update({ split_between: newSplit })
        .eq("id", id);
      if (error) throw error;
      dynaura.update(notificationId, { progress: 0.72 });
      await refreshExpenses();
      dynaura.update(notificationId, {
        type: "success",
        title: "อัปเดตการหารสำเร็จ",
        description: "ยอดแบ่งรายการได้รับการอัปเดต",
        progress: undefined,
        duration: 3000,
      });
      return true;
    } catch (error) {
      dynaura.update(notificationId, {
        type: "error",
        title: "อัปเดตไม่สำเร็จ",
        description: error.message,
        progress: undefined,
        duration: 5000,
      });
      return false;
    } finally {
      setPendingSplitId(null);
    }
  };

  const cancelEdit = () => {
    setEditId(null);
    setDate(getTodayISO());
    setItem("");
    setAmount("");
    setCategory(CATEGORIES[0]);
    setOdometerStart("");
    setOdometerEnd("");
    setDidEditOdometerStart(false);
    setDidEditDraftSplit(false);
  };

  const handleDelete = async (id) => {
    if (pendingDeleteId) return;
    if (isOffline)
      return dynaura.warning({
        title: "ออฟไลน์",
        description: "ไม่สามารถลบรายการขณะออฟไลน์ได้",
        duration: 4000,
      });
    const r = await Swal.fire({
      ...SWAL_THEME,
      title: "ยืนยันการลบ?",
      showCancelButton: true,
      confirmButtonText: "ลบ",
      customClass: {
        ...SWAL_THEME.customClass,
        confirmButton:
          "bg-red-600 text-white font-bold px-5 py-2.5 rounded-lg hover:bg-red-500 transition text-sm cursor-pointer",
      },
    });
    if (r.isConfirmed) {
      setPendingDeleteId(id);
      const notificationId = dynaura.loading({
        id: `delete-expense-${id}-${Date.now()}`,
        title: "กำลังลบรายการ",
        description: "กำลังอัปเดตรายจ่ายและยอดคงเหลือ",
        progress: 0.15,
      });
      try {
        const { error } = await supabase
          .from("trip_expenses")
          .delete()
          .eq("id", id);
        if (error) throw error;
        dynaura.update(notificationId, { progress: 0.72 });
        await refreshExpenses();
        dynaura.update(notificationId, {
          type: "success",
          title: "ลบรายการแล้ว",
          description: "รายจ่ายและยอดคงเหลือถูกอัปเดต",
          progress: undefined,
          duration: 3000,
        });
      } catch (error) {
        dynaura.update(notificationId, {
          type: "error",
          title: "ลบไม่สำเร็จ",
          description: error.message,
          progress: undefined,
          duration: 5000,
        });
      } finally {
        setPendingDeleteId(null);
      }
    }
  };

  const handleUndoHistory = async (id) => {
    if (pendingHistoryId) return;
    if (isOffline)
      return dynaura.warning({
        title: "ออฟไลน์",
        description: "ไม่สามารถกู้คืนยอดขณะออฟไลน์ได้",
        duration: 4000,
      });
    setPendingHistoryId(id);
    const notificationId = dynaura.loading({
      id: `undo-history-${id}-${Date.now()}`,
      title: "กำลังกู้คืนยอด",
      description: "กำลังลบประวัติการเคลียร์ยอด",
      progress: 0.15,
    });
    try {
      const { error } = await supabase
        .from("trip_expenses")
        .delete()
        .eq("id", id);
      if (error) throw error;
      dynaura.update(notificationId, { progress: 0.72 });
      await refreshExpenses();
      dynaura.update(notificationId, {
        type: "success",
        title: "กู้คืนยอดเรียบร้อย",
        description: "ยอดค้างชำระถูกคำนวณใหม่",
        progress: undefined,
        duration: 3000,
      });
    } catch (error) {
      dynaura.update(notificationId, {
        type: "error",
        title: "กู้คืนไม่สำเร็จ",
        description: error.message,
        progress: undefined,
        duration: 5000,
      });
    } finally {
      setPendingHistoryId(null);
    }
  };

  const handleSettleDebt = async (from, to, settleAmount) => {
    if (settlementSubmitLock.current) return;
    if (isOffline)
      return Swal.fire({
        ...SWAL_THEME,
        icon: "warning",
        title: "ออฟไลน์",
        text: "ไม่สามารถทำรายการขณะออฟไลน์ได้",
      });
    const today = new Date();
    const formattedDate = `${String(today.getDate()).padStart(2, "0")}/${String(today.getMonth() + 1).padStart(2, "0")}/${today.getFullYear()}`;
    const newHistory = {
      expense_date: formattedDate,
      item_name: `[เคลียร์ยอด] ${from} -> ${to}`,
      amount: settleAmount,
      category: HISTORY_CATEGORY,
      paid_by: from,
      split_between: [from], // 👈 🧠 เพิ่มบรรทัดนี้เข้ามา! เพื่อแก้บั๊กไม่ให้ฐานข้อมูลเตือน Error
    };

    settlementSubmitLock.current = true;
    setPendingSettlement(`${from}-${to}`);
    const notificationId = dynaura.loading({
      id: `settle-${from}-${to}-${Date.now()}`,
      title: "กำลังบันทึกประวัติการโอน",
      description: `${from} → ${to}`,
      progress: 0.15,
    });
    try {
      const { error } = await supabase
        .from("trip_expenses")
        .insert([newHistory]);
      if (error) throw error;
      dynaura.update(notificationId, { progress: 0.72 });
      setActivePayment(null);
      await refreshExpenses();
      dynaura.update(notificationId, {
        type: "success",
        title: "บันทึกประวัติการโอนแล้ว",
        description: `฿${formatMoney(settleAmount)} จาก ${from} ถึง ${to}`,
        progress: undefined,
        duration: 4000,
        confetti: true,
      });
    } catch (error) {
      dynaura.update(notificationId, {
        type: "error",
        title: "บันทึกไม่สำเร็จ",
        description: error.message,
        progress: undefined,
        duration: 5000,
      });
    } finally {
      settlementSubmitLock.current = false;
      setPendingSettlement(null);
    }
  };

  // ปิดการทำงานกดค้างเพื่อเลื่อนไปด้านบน
  // const startPressTimer = () => {
  //   hasScrolled.current = false;
  //   setIsPressing(true);
  //   pressTimer.current = setTimeout(() => {
  //     window.scrollTo({ top: 0, behavior: "smooth" });
  //     hasScrolled.current = true;
  //     setIsPressing(false);
  //   }, 400);
  // };
  // const cancelPressTimer = () => {
  //   clearTimeout(pressTimer.current);
  //   setIsPressing(false);
  // };

  const regularExpenses = expenses.filter(
    (e) => e.category !== HISTORY_CATEGORY,
  );
  const historyExpenses = expenses.filter(
    (e) => e.category === HISTORY_CATEGORY,
  );
  const groupedExpenses = useMemo(
    () =>
      regularExpenses.reduce((groups, expense) => {
        (groups[expense.expense_date] ||= []).push(expense);
        return groups;
      }, {}),
    [regularExpenses],
  );
  const totalExpense = useMemo(
    () => regularExpenses.reduce((sum, e) => sum + Number(e.amount), 0),
    [regularExpenses],
  );

  const settlements = useSettlements(expenses, MEMBERS);

  // หน้าจอตอนกำลังโหลดข้อมูลตอนเปิดแอปครั้งแรก
  // if (loading) {
  //   return (
  //     <div className="flex min-h-screen flex-col items-center justify-center bg-gray-950 p-6 text-center text-gray-100">
  //       <Loader2 size={48} className="mb-4 animate-spin text-lime-400" />
  //       <p className="text-sm font-bold text-gray-400">
  //         กำลังเชื่อมต่อฐานข้อมูล...
  //       </p>
  //     </div>
  //   );
  // }

  return (
    <div
      className="relative min-h-dvh bg-gray-950 pb-24 font-sans text-gray-100"
      onClick={handleAppClick}
    >
      <Toaster position="top" offset={0} haptics />
      <style>{`
        @keyframes drawRing { 0% { stroke-dashoffset: 176; } 100% { stroke-dashoffset: 0; } }
        .animate-draw-fast { animation: drawRing 0.4s linear forwards; }

        /*
         * DYNAURA_TOASTER_TOP:
         * แสดงใต้ safe area แล้วปรับระยะเพิ่มเติมจากขอบบนได้ที่เลข px
         * เพิ่มเลข px เพื่อขยับลง; ลดเลขเพื่อขยับขึ้น
         */
        .dynaura-root[data-y="top"] {
          top: max(11px, calc(env(safe-area-inset-top, 0px) - 48px));
        }

        /*
         * DYNAURA_TOASTER_SIZE:
         * ปรับความกว้างสูงสุดได้ที่ min(380px, 92vw)
         * เพิ่ม 380px เป็นขนาดกว้างขึ้น ลดลงเพื่อเล็กลง

        max-width: min(350px, 88vw);
        }
        .dynaura-island[data-expanded="true"] .dynaura-inner {
          min-width: min(320px, 88vw);

      <div className="fixed right-3 top-[calc(env(safe-area-inset-top,0px)+1rem)] z-70 rounded-md  px-1.5 py-1 pointer-events-none select-none text-[10px] text-gray-700 opacity-50 shadow-lg">
        {version_number}
      </div>

      {isOffline && (
        <div
          role="status"
          className="relative z-40 flex items-center justify-center gap-2 border-b border-amber-500/40 bg-amber-500/95 px-3 py-2 text-center text-xs font-bold text-gray-950"
        >
          <WifiOff size={14} />
          <span>ออฟไลน์อยู่ — ดูข้อมูลล่าสุดจากเครื่อง</span>
        </div>
      )}

      <TripHeader
        totalExpense={totalExpense}
        isSyncing={loading || isSyncing}
        onRefresh={() => {
          refreshMembers();
          refreshExpenses();
        }}
        MEMBERS={MEMBERS}
        isOffline={isOffline}
        onOpenMemberModal={() => setShowMemberModal(true)}
      />

      <main className="mx-auto grid max-w-6xl grid-cols-1 items-start gap-6 p-4 lg:grid-cols-3">

          date={date}
          setDate={setDate}
          item={item}
          setItem={setItem}
          amount={amount}
          setAmount={setAmount}
          category={category}
          setCategory={setCategory}
          paidBy={paidBy}
          setPaidBy={(value) => {
            setPaidBy(value);
            localStorage.setItem("defaultPaidBy", value);
            if (!editId && !didEditDraftSplit) setDraftSplitBetween([value]);
          }}
          members={MEMBERS}
          odometerStart={odometerStart}
          setOdometerStart={setOdometerStart}
          setDidEditOdometerStart={setDidEditOdometerStart}
          odometerEnd={odometerEnd}
          setOdometerEnd={setOdometerEnd}
          draftSplitBetween={draftSplitBetween}
          onOpenSplit={() =>
            setActiveSplitExpense({
              __draft: true,
              item_name: item.trim() || category,
              amount: Number(amount) || 0,
              paid_by: paidBy,
              split_between: draftSplitBetween,
            })
          }
          isSubmitting={isSubmittingExpense}
          onSubmit={handleAddExpense}
          onCancel={cancelEdit}
          onAddMember={() => setShowMemberModal(true)}
        />

        <div className="lg:col-span-2" onClick={(e) => e.stopPropagation()}>
          <ExpenseList
            expenses={regularExpenses}
            groupedExpenses={groupedExpenses}
            loading={false}
            isExpanded={isListExpanded}
            collapsedDates={collapsedDates}
            onToggleExpanded={() => setIsListExpanded(!isListExpanded)}
            onToggleDate={(val) =>
              setCollapsedDates((d) => ({ ...d, [val]: !d[val] }))
            }
            onEdit={(exp) => {
              setEditId(exp.id);
              setDate(parseForUI(exp.expense_date));
              setPaidBy(exp.paid_by);
              setItem(exp.item_name);
              setAmount(exp.amount);
              setCategory(exp.category);
              setOdometerStart(exp.odometer_start || "");
              setOdometerEnd(exp.odometer_end || "");
              window.scrollTo({ top: 0, behavior: "smooth" });
            }}
            onDelete={handleDelete}
            onOpenSplitModal={setActiveSplitExpense}
            pendingDeleteId={pendingDeleteId}
            pendingSplitId={pendingSplitId}
            swipedId={swipedId}
            setSwipedId={setSwipedId}
            highlightedId={highlightedId}
            MEMBERS={MEMBERS}
          />
        </div>

        <SettlementSection
          settlements={settlements}
          onOpenPayment={setActivePayment}
          onOpenDetails={setActiveSettlementDetails}
          onOpenHistory={() => setShowHistory(true)}
          pendingSettlement={pendingSettlement}
          isOffline={isOffline}
        />
      </main>

      <ThemeMenu />

      {showMemberModal && (
        <MemberModal
          members={MEMBERS}
          balances={settlements.balances}
          memberDetails={memberDetails}
          onClose={() => setShowMemberModal(false)}
          onAdd={handleAddMember}
          onRemove={handleRemoveMember}
          onUpdateDetails={handleUpdateMemberDetails}
          pendingMemberAction={pendingMemberAction}
          isOffline={isOffline}
        />
      )}
      {activePayment && (
        <PaymentModal
          from={activePayment.from}
          to={activePayment.to}
          amount={activePayment.amount}
          memberDetails={memberDetails}
          onClose={() => setActivePayment(null)}
          onUpdateDetails={handleUpdateMemberDetails}
          onConfirm={handleSettleDebt}
          isSubmitting={Boolean(pendingSettlement)}
        />
      )}
      {activeSettlementDetails && (
        <SettlementDetailsModal
          transfer={activeSettlementDetails}
          onClose={() => setActiveSettlementDetails(null)}
        />
      )}
      {activeSplitExpense && (
        <SplitModal
          expense={activeSplitExpense}
          members={MEMBERS}
          onClose={() => setActiveSplitExpense(null)}
          onSave={handleUpdateSplit}
        />
      )}
      {showHistory && (
        <HistoryModal
          expenses={historyExpenses}
          onClose={() => setShowHistory(false)}
          onUndo={handleUndoHistory}
          pendingHistoryId={pendingHistoryId}
          isOffline={isOffline}
        />
      )}
    </div>
  );
}

import { Edit2, Trash2, Users } from "lucide-react";
import { useEffect, useRef } from "react";
import { formatMoney } from "../utils/trip";
import useSwipeDrag from "../hooks/useSwipeDrag";
import { dynaura } from "dynaura";

export default function ExpenseItem({
  expense,
  MEMBERS,
  swipedId,
  setSwipedId,
  onEdit,
  onDelete,
  onOpenSplitModal,
  isLast,
  isHighlighted, //  รับ State การ Highlight
  isPendingDelete,
  isPendingSplit,
  isOffline,
}) {
  const validSplit = expense.split_between?.filter(m => MEMBERS.includes(m)) || [];
  const itemRef = useRef(null); //  Ref สำหรับเลื่อนหน้าจอ
  const isOpen = swipedId === expense.id;
  const isPendingSync = expense.isPendingSync === true;
  const canInteract = !isOffline && !isPendingSync;
  const {
    drag,
    currentOffset,
    isSwiped,
    consumeSuppressedClick,
    touchHandlers,
    mouseHandlers,
  } = useSwipeDrag({
    id: expense.id,
    isOpen,
    setOpen: setSwipedId,
    canDrag: canInteract,
  });

  const createdTime = expense.created_at
    ? new Date(expense.created_at).toLocaleTimeString([], {
        hour: "2-digit",
        minute: "2-digit",
        hour12: false,
      })
    : "";

  const warnOffline = () =>
    dynaura.warning({
      title: "ออฟไลน์",
      description: "ไม่สามารถแก้ไขหรือลบรายการขณะออฟไลน์ได้",
      duration: 4000,
    });

  //  สั่ง Auto-scroll มาที่ Item นี้เมื่อโดน Highlight
  useEffect(() => {
    if (isHighlighted && itemRef.current) {
      itemRef.current.scrollIntoView({
        behavior: "smooth",
        block: "center", // ให้กล่องรายการอยู่ตรงกลางจอพอดี
      });
    }
  }, [isHighlighted]);

  const splitList = validSplit.length > 0 
  ? validSplit
  : (MEMBERS.includes(expense.paid_by) ? [expense.paid_by] : [MEMBERS[0]]); // ถ้าไม่มีใครอยู่ใน split_between ให้ใช้ paid_by หรือสมาชิกคนแรกแทน
  const isGas = expense.category === "ค่าน้ำมัน";
  const distance =
    isGas && expense.odometer_start && expense.odometer_end
      ? Number(expense.odometer_end) - Number(expense.odometer_start)
      : 0;


  return (
    <div
      ref={itemRef} //  ผูก Ref เพื่อดึงหน้าจอ
      className={`relative bg-gray-900 ${!isLast ? "border-b border-gray-800" : ""}`}
    >
      <div className="absolute inset-y-0 right-3 flex items-center justify-center">
        <button
          onClick={() => {
            if (!canInteract) {
              warnOffline();
              return;
            }
            onDelete(expense.id);
            setSwipedId(null);
          }}
          disabled={isPendingDelete || !canInteract}
          className={`flex h-11 w-11 items-center justify-center rounded-full text-white shadow-sm transition ${
            isPendingDelete || !canInteract
              ? "cursor-not-allowed bg-gray-500 shimmer"
              : "bg-red-500 hover:bg-red-600"
          }`}
          style={{
            opacity: Math.min(1, -currentOffset / 40),
            transform: `scale(${Math.min(1.2, 0.2 + (-currentOffset / 70) * 0.8)})`,
            transition: drag.active
              ? "none"
              : "all 0.3s cubic-bezier(0.4, 0, 0.2, 1)",
          }}
        >
          <Trash2 size={20} />
        </button>
      </div>

      <div
        //  สลับ Class การสว่าง (Highlight สีเหลือง) หาก isHighlighted เป็น true
        className={`relative z-10 flex cursor-grab touch-pan-y items-center justify-between p-3 pl-8 active:cursor-grabbing transition-colors duration-700 ${
          isHighlighted ? "bg-yellow-200/30 shadow-[inset_4px_0_0_#c9c9af]" : "bg-gray-800"
        }`}
        style={{
          transform: `translate3d(${currentOffset}px, 0, 0)`,
          transition: drag.active
            ? "none"
            : "transform 0.5s cubic-bezier(0.4, 0, 0.2, 1), background-color 0.5s ease-out, box-shadow 0.5s ease-out, border-radius 0.3s cubic-bezier(0.4, 0, 0.2, 1) 0.1s",
          borderTopRightRadius: isSwiped ? "24px" : "0px",
          borderBottomRightRadius: isSwiped ? "24px" : "0px",
        }}
        {...touchHandlers}
        {...mouseHandlers}
        onClick={(e) => {
          if (consumeSuppressedClick()) {
            e.stopPropagation();
            return;
          }
          setSwipedId(null);
        }}
      >
        <div className="pointer-events-none flex-1">
          <p className="mb-0.5 text-xs text-gray-400">
            จ่ายโดย:{" "}
            <span className="font-bold text-lime-400">{expense.paid_by}</span>
            {createdTime && (
              <span className="ml-1 text-[11px] font-normal text-gray-400">
                {createdTime}
              </span>
            )}
            {isPendingSync && (
              <span className="ml-2 text-xs font-bold text-amber-400">
                รอซิงค์
              </span>
            )}
          </p>
          <p className="font-medium text-gray-100">{expense.item_name}</p>
          <p className="text-xs text-gray-400">
            {expense.category}{" "}
            {splitList.length > 0 && (
              <span className="text-[10px] text-sky-400">
                {" "}
                (หาร {splitList.length} คน)
              </span>
            )}
          </p>
          {distance > 0 && (
            <p className="mt-0.5 text-[11px] font-medium text-amber-400">
              ไมล์ {expense.odometer_start} - {expense.odometer_end} ({distance}{" "}
              กม.)
            </p>
          )}
        </div>
        <div className="flex items-center gap-3">
          <p className="pointer-events-none font-bold text-gray-100">
            ฿{formatMoney(expense.amount)}
          </p>
          <div className="relative z-20 flex items-center gap-2">
            <button
              disabled={isPendingSplit || !canInteract}
              onClick={(e) => {
                e.stopPropagation();
                if (!canInteract) {
                  warnOffline();
                  return;
                }
                onOpenSplitModal(expense);
              }}
              className={`flex h-10 w-10 items-center justify-center rounded-full border text-gray-400 transition ${
                isPendingSplit || !canInteract
                  ? "cursor-not-allowed border-gray-500 bg-gray-500 shimmer"
                  : "border-gray-600 hover:border-sky-400 hover:bg-sky-400/10 hover:text-sky-400"
              }`}
              title="เลือกคนหาร"
              aria-label="เลือกคนหาร"
            >
              <Users size={19} />
            </button>
            <button
              disabled={!canInteract}
              onClick={(e) => {
                e.stopPropagation();
                if (!canInteract) {
                  warnOffline();
                  return;
                }
                onEdit(expense);
                setSwipedId(null);
              }}
              className={`flex h-10 w-10 items-center justify-center rounded-full text-gray-400 transition ${
                !canInteract
                  ? "opacity-40 cursor-not-allowed"
                  : "hover:bg-lime-400/10 hover:text-lime-400"
              }`}
              title="แก้ไขรายการ"
              aria-label="แก้ไขรายการ"
            >
              <Edit2 size={20} />
            </button>
          </div>
        </div>
      </div>
    </div>
  );
}

import { X } from "lucide-react";
import { useEffect, useState } from "react";
import { formatMoney } from "../utils/trip";
import ModalShell from "./ModalShell";

export default function SplitModal({ expense, members, onClose, onSave }) {
  const [isSaving, setIsSaving] = useState(false);
  const [selectedSplit, setSelectedSplit] = useState(() => {
    const rawSplit =
      expense?.split_between?.length > 0
        ? expense.split_between
        : [expense.paid_by];
    return rawSplit.filter((m) => members.includes(m));
  });

  useEffect(() => {
    const handleEsc = (e) => {
      if (e.key === "Escape") onClose();
    };
    document.addEventListener("keydown", handleEsc);
    return () => document.removeEventListener("keydown", handleEsc);
  }, [onClose]);

  const handleToggle = (member) =>
    setSelectedSplit((prev) =>
      prev.includes(member)
        ? prev.filter((m) => m !== member)
        : [...prev, member],
    );
  const isAllSelected =
    selectedSplit.length === members.length && members.length > 0;
  const handleToggleAll = () =>
    isAllSelected ? setSelectedSplit([]) : setSelectedSplit([...members]);

  return (
    <ModalShell
      backdropClassName="bg-gray-950/70"
      panelClassName="max-w-sm"
      onClose={onClose}
    >
        <div className="flex items-center justify-between border-b border-gray-800 bg-gray-800 p-4 text-lime-400">
          <div>
            <h3 className="font-bold text-base text-gray-100">
              ระบุผู้หารรายการ
            </h3>
            <p className="text-xs text-lime-400 font-medium">
              {expense.item_name} (฿{formatMoney(expense.amount)})
            </p>
          </div>
          <button
            onClick={onClose}
            className="text-gray-400 hover:text-gray-200"
          >
            <X size={20} />
          </button>
        </div>

        <div className="p-4">
          <div className="mb-3 flex items-center justify-between">
            <p className="text-xs font-bold text-gray-400">เลือกคนที่ร่วมหาร</p>
            <label className="flex cursor-pointer items-center gap-1.5 text-xs font-semibold text-lime-400 hover:text-lime-300">
              <input
                type="checkbox"
                checked={isAllSelected}
                onChange={handleToggleAll}
                className="split-check h-3.5 w-3.5 cursor-pointer"
              />
              <span>เลือกทุกคน</span>
            </label>
          </div>
          <div className="mb-4 max-h-56 space-y-1.5 overflow-y-auto pr-1">
            {members.map((member) => (
              <label
                key={member}
                className="flex cursor-pointer items-center justify-between rounded-lg border border-gray-800 bg-gray-800/60 p-3 transition hover:bg-gray-750"
              >
                <span className="font-medium text-gray-100 text-sm">
                  {member}
                </span>
                <input
                  type="checkbox"
                  checked={selectedSplit.includes(member)}
                  onChange={() => handleToggle(member)}
                  className="split-check h-4 w-4 cursor-pointer"
                />
              </label>
            ))}
          </div>
          <div className="flex gap-2">
            <button
              type="button"
              onClick={onClose}
              className="flex-1 rounded-lg border border-gray-700 bg-gray-800 py-2.5 text-xs font-semibold text-gray-300 transition hover:bg-gray-700"
            >
              ยกเลิก
            </button>
            <button
              type="button"
              disabled={isSaving}
              onClick={async () => {
                setIsSaving(true);
                const success = expense.__draft
                  ? await onSave(null, selectedSplit)
                  : await onSave(expense.id, selectedSplit);
                setIsSaving(false);
                if (success) onClose();
              }}
              className={`flex-1 rounded-lg py-2.5 text-xs font-bold text-gray-950 transition ${
                isSaving
                  ? "cursor-not-allowed bg-gray-500 shimmer"

import { createContext, useContext, useEffect, useState } from "react";

const ThemeContext = createContext(null);

export function ThemeProvider({ children }) {
  const [theme, setTheme] = useState(
    () => localStorage.getItem("tripTheme") || "pink",
  );

  useEffect(() => {
    document.documentElement.dataset.theme = theme;
    localStorage.setItem("tripTheme", theme);

    const themeColors = {
      dark: "#111827",
      pink: "#fec7d7",
      light: "#55423d",
      purple: "#faeee7",
    };
    const themeColor = themeColors[theme] || themeColors.dark;

    let themeMeta = document.querySelector('meta[name="theme-color"]');
    if (!themeMeta) {
      themeMeta = document.createElement("meta");
      themeMeta.name = "theme-color";
      document.head.appendChild(themeMeta);
    }
    themeMeta.content = themeColor;

    const statusBarMeta = document.querySelector(
      'meta[name="apple-mobile-web-app-status-bar-style"]',
    );
    if (statusBarMeta) {
      statusBarMeta.content = theme === "dark" ? "black" : "default";
    }
  }, [theme]);

  return (
    <ThemeContext.Provider value={{ theme, setTheme }}>
      {children}
    </ThemeContext.Provider>
  );
}

export function useTheme() {
  const context = useContext(ThemeContext);
  if (!context) throw new Error("useTheme must be used within ThemeProvider");
  return context;
}

===== src/hooks/useSettlements.js =====
import { useMemo } from "react";
import { HISTORY_CATEGORY } from "../data/constants";

const roundMoney = (value) => Math.round((Number(value) + Number.EPSILON) * 100) / 100;

const parseExpenseTimestamp = (expense) => {
  let dateScore = 0;
  if (typeof expense.expense_date === "string" && expense.expense_date.includes("/")) {
    const [day, month, year] = expense.expense_date.split("/").map(Number);
    if (day && month && year) {
      dateScore = new Date(year, month - 1, day).getTime();
    }
  }

  const createdScore = expense.created_at ? new Date(expense.created_at).getTime() : 0;
  const fallbackId = Number(expense.id) || 0;

  return {
    dateScore: Number.isFinite(dateScore) ? dateScore : 0,
    timeScore: Number.isFinite(createdScore) && createdScore > 0 ? createdScore : fallbackId,
  };
};

const compareNewestFirst = (a, b) => {
  const timeA = parseExpenseTimestamp(a);
  const timeB = parseExpenseTimestamp(b);
  if (timeB.dateScore !== timeA.dateScore) {
    return timeB.dateScore - timeA.dateScore;
  }
  return timeB.timeScore - timeA.timeScore;
};

const getValidSplitters = (expense, members) => {
  const configuredSplit = Array.isArray(expense.split_between)
    ? expense.split_between
    : [];
  const validSplit = configuredSplit.filter((member) => members.includes(member));
  if (validSplit.length > 0) return validSplit;
  if (members.includes(expense.paid_by)) return [expense.paid_by];
  return members.length > 0 ? [members[0]] : [];
};

export default function useSettlements(expenses, members) {
  return useMemo(() => {
    const regularExpenses = expenses.filter(
      (expense) => expense.category !== HISTORY_CATEGORY,
    );
    const historyExpenses = expenses.filter(
      (expense) => expense.category === HISTORY_CATEGORY,
    );

    const balanceMap = new Map(members.map((member) => [member, 0]));
    const transfers = [];

    for (let i = 0; i < members.length; i += 1) {
      for (let j = i + 1; j < members.length; j += 1) {
        const personA = members[i];
        const personB = members[j];

        const paidByAForB = [];
        const paidByBForA = [];

        regularExpenses.forEach((expense) => {
          const splitters = getValidSplitters(expense, members);
          if (splitters.length === 0) return;

          const shareAmount = roundMoney(Number(expense.amount) / splitters.length);
          if (shareAmount <= 0) return;

          if (expense.paid_by === personA && splitters.includes(personB)) {
            paidByAForB.push({
              ...expense,
              splitCount: splitters.length,
              shareAmount,
            });
          } else if (expense.paid_by === personB && splitters.includes(personA)) {
            paidByBForA.push({
              ...expense,
              splitCount: splitters.length,
              shareAmount,
            });
          }
        });

        const totalAForB = roundMoney(
          paidByAForB.reduce((sum, item) => sum + item.shareAmount, 0),
        );
        const totalBForA = roundMoney(
          paidByBForA.reduce((sum, item) => sum + item.shareAmount, 0),
        );

        const diff = roundMoney(totalAForB - totalBForA);
        if (Math.abs(diff) <= 0.005) continue;

        const isBOwesA = diff > 0;
        const from = isBOwesA ? personB : personA;
        const to = isBOwesA ? personA : personB;
        const amount = Math.abs(diff);

        const positiveItems = (isBOwesA ? paidByAForB : paidByBForA).sort(
          compareNewestFirst,
        );
        const negativeItems = (isBOwesA ? paidByBForA : paidByAForB).sort(
          compareNewestFirst,
        );

        const positiveTotal = isBOwesA ? totalAForB : totalBForA;
        const negativeTotal = isBOwesA ? totalBForA : totalAForB;

        const settledHistory = historyExpenses
          .filter(
            (item) =>
              item.item_name.includes(`${from} -> ${to}`) ||
              item.item_name.includes(`${to} -> ${from}`),
          )
          .sort(compareNewestFirst);

        balanceMap.set(from, roundMoney((balanceMap.get(from) || 0) - amount));
        balanceMap.set(to, roundMoney((balanceMap.get(to) || 0) + amount));

        transfers.push({
          from,
          to,
          amount,
          positiveItems,
          negativeItems,
          positiveTotal,
          negativeTotal,
          settledHistory,
        });
      }
    }

    const balances = members.map((member) => ({
      member,
      balance: balanceMap.get(member) || 0,
    }));

    return { balances, transfers };
  }, [expenses, members]);
}
===== src/hooks/useSwipeDrag.js =====
import { useRef, useState } from "react";

export default function useSwipeDrag({
  id,
  isOpen,
  setOpen,
  canDrag = true,
}) {
  const [drag, setDrag] = useState({
    active: false,
    startX: 0,
    startY: 0,
    offset: 0,
    scrolling: false,
  });
  const isMouse = useRef(false);
  const suppressClick = useRef(false);

  let currentOffset = isOpen ? -70 : 0;
  if (drag.active) {
    currentOffset = (isOpen ? -70 : 0) + drag.offset;
    currentOffset = Math.max(-90, Math.min(0, currentOffset));
  }

  const startDrag = (clientX, clientY) => {
    if (!canDrag) return;
    setDrag({
      active: true,
      startX: clientX,
      startY: clientY,
      offset: 0,
      scrolling: false,
    });
    if (!isOpen) setOpen(null);
  };

  const moveDrag = (clientX, clientY) => {
    if (!drag.active) return;
    const diffX = clientX - drag.startX;
    const diffY = Math.abs(clientY - drag.startY);

    if (!drag.scrolling && diffY > 15 && Math.abs(diffX) < 15) {
      setDrag((previous) => ({ ...previous, scrolling: true }));
      return;
    }
    if (!drag.scrolling) {
      setDrag((previous) => ({ ...previous, offset: diffX }));
    }
  };

  const endDrag = () => {
    if (!drag.active) return;
    suppressClick.current = Math.abs(drag.offset) > 5;
    isMouse.current = false;

    if (!drag.scrolling) {
      const baseOffset = isOpen ? -70 : 0;
      if (baseOffset + drag.offset < -35) setOpen(id);
      else setOpen(null);
    }

    setDrag({
      active: false,
      startX: 0,
      startY: 0,
      offset: 0,
      scrolling: false,
    });
  };

  const touchHandlers = {
    onTouchStart: (event) =>
      startDrag(
        event.targetTouches[0].clientX,
        event.targetTouches[0].clientY,
      ),
    onTouchMove: (event) =>
      moveDrag(
        event.targetTouches[0].clientX,
        event.targetTouches[0].clientY,
      ),
    onTouchEnd: endDrag,
  };

  const mouseHandlers = {
    onMouseDown: (event) => {
      isMouse.current = true;
      startDrag(event.clientX, event.clientY);
    },
    onMouseMove: (event) => {
      if (isMouse.current) moveDrag(event.clientX, event.clientY);
    },
    onMouseUp: () => {
      if (isMouse.current) endDrag();
    },
    onMouseLeave: () => {
      if (isMouse.current) endDrag();
    },
  };

  const consumeSuppressedClick = () => {
    if (!suppressClick.current) return false;
    suppressClick.current = false;
    return true;
  };

  return {
    drag,
    currentOffset,
    isSwiped: currentOffset < -5,
    consumeSuppressedClick,
    touchHandlers,
    mouseHandlers,
  };
}

===== src/hooks/useTripData.js =====
import { useCallback, useEffect, useState } from "react";
import { getOfflineQueue, getStoredMembers } from "../utils/trip";
import { supabase } from "../services/supabase";

export default function useTripData(onMembersUpdated, onMemberDetailsUpdated) {
  const [members, setMembers] = useState(getStoredMembers);
  const [expenses, setExpenses] = useState([]);
  const [isOffline, setIsOffline] = useState(!navigator.onLine);
  const [loading, setLoading] = useState(true);
  const [isSyncing, setIsSyncing] = useState(false);

  const refreshMembers = useCallback(async () => {
    if (!navigator.onLine) return;

    const { data } = await supabase
      .from("trip_members")
      .select("name, promptpay, bank_id, bank_acc")
      .order("created_at", { ascending: true });

    if (data && data.length > 0) {
      const fetchedMembers = data.map((member) => member.name);
      const fetchedDetails = data.reduce((details, member) => {
        details[member.name] = {
          promptpay: member.promptpay || "",
          bankId: member.bank_id || "kbank",
          bankAcc: member.bank_acc || "",
        };
        return details;
      }, {});
      setMembers(fetchedMembers);
      localStorage.setItem("tripMembers", JSON.stringify(fetchedMembers));
      localStorage.setItem("memberDetails", JSON.stringify(fetchedDetails));
      onMembersUpdated(fetchedMembers);
      if (onMemberDetailsUpdated) onMemberDetailsUpdated(fetchedDetails);
    }
  }, [onMembersUpdated, onMemberDetailsUpdated]);

  const refreshExpenses = useCallback(async () => {
    if (!navigator.onLine) {
      const cachedData = localStorage.getItem("tripExpensesCache");
      const cachedExpenses = cachedData ? JSON.parse(cachedData) : [];
      const queuedExpenses = getOfflineQueue();
      const mergedById = new Map();
      [...cachedExpenses, ...queuedExpenses].forEach((expense) => {
        mergedById.set(expense.id, expense);
      });
      const mergedExpenses = [...mergedById.values()].sort(
        (a, b) => new Date(b.created_at || 0) - new Date(a.created_at || 0),
      );
      setExpenses(mergedExpenses);
      localStorage.setItem("tripExpensesCache", JSON.stringify(mergedExpenses));
      return;
    }

    setIsSyncing(true);
    try {
      const { data, error } = await supabase
        .from("trip_expenses")
        .select("*")
        .order("created_at", { ascending: false });

      if (error) throw error;

      if (data) {
        const pendingQueue = getOfflineQueue();
        const mergedById = new Map();
        [...data, ...pendingQueue].forEach((expense) => {
          mergedById.set(expense.id, expense);
        });
        const mergedExpenses = [...mergedById.values()].sort(
          (a, b) =>
            new Date(b.created_at || 0) - new Date(a.created_at || 0),
        );
        setExpenses(mergedExpenses);
        localStorage.setItem(
          "tripExpensesCache",
          JSON.stringify(mergedExpenses),
        );
      }
    } catch {
      const cachedData = localStorage.getItem("tripExpensesCache");
      if (cachedData) setExpenses(JSON.parse(cachedData));
    } finally {
      setTimeout(() => setIsSyncing(false), 500);
    }
  }, []);

  useEffect(() => {
    setIsOffline(!navigator.onLine);

    const handleOnline = () => {
      setIsOffline(false);
      Promise.all([refreshMembers(), refreshExpenses()]).then(() =>
        setLoading(false),
      );
    };
    const handleOffline = () => setIsOffline(true);

    window.addEventListener("online", handleOnline);
    window.addEventListener("offline", handleOffline);

    Promise.all([refreshMembers(), refreshExpenses()]).then(() =>
      setLoading(false),
    );

    const expenseSub = supabase
      .channel("trip_expenses_changes")
      .on(
        "postgres_changes",
        { event: "*", schema: "public", table: "trip_expenses" },
        refreshExpenses,
      )
      .subscribe();
    const memberSub = supabase
      .channel("trip_members_changes")
      .on(
        "postgres_changes",
        { event: "*", schema: "public", table: "trip_members" },
        refreshMembers,
      )
      .subscribe();

    return () => {
      window.removeEventListener("online", handleOnline);
      window.removeEventListener("offline", handleOffline);
      supabase.removeChannel(expenseSub);
      supabase.removeChannel(memberSub);
    };
  }, [refreshExpenses, refreshMembers]);

  return {
    members,
    setMembers,
    expenses,
    setExpenses,
    isOffline,
    loading,
    isSyncing,
    refreshMembers,
    refreshExpenses,
  };
}

import { dynaura } from "dynaura";
import { DEFAULT_MEMBERS } from "../data/constants";

export const getTodayISO = () => new Date().toLocaleDateString("en-CA");

export const formatForDB = (isoDate) => {
  if (!isoDate || !isoDate.includes("-")) return isoDate;
  const [y, m, d] = isoDate.split("-");
  return `${d}/${m}/${y}`;
};

export const parseForUI = (thDate) => {
  if (!thDate || !thDate.includes("/")) return getTodayISO();
  const [d, m, y] = thDate.split("/");
  return `${y}-${m.padStart(2, "0")}-${d.padStart(2, "0")}`;
};

export const getStoredMembers = () => {
  try {
    const saved = JSON.parse(localStorage.getItem("tripMembers") || "null");
    if (Array.isArray(saved) && saved.length > 0) return saved;
  } catch {
    // Ignore malformed local data and use the defaults.
  }
  return DEFAULT_MEMBERS;
};

export const getStoredMemberDetails = () => {
  try {
    const saved = JSON.parse(localStorage.getItem("memberDetails") || "null");
    if (saved) return saved;
  } catch {
    // Ignore malformed local data and use an empty details object.
  }
  return {};
};

export const formatMoney = (value) =>
  Number(value).toLocaleString(undefined, {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  });

export const normalizePromptPay = (value) => {
  let normalized = String(value || "").trim().replace(/[\s-]/g, "");
  if (normalized.startsWith("+66")) {
    normalized = `0${normalized.slice(3)}`;
  } else if (normalized.startsWith("66") && normalized.length === 11) {
    normalized = `0${normalized.slice(2)}`;
  }

  return normalized;
};

export const validatePromptPay = (value) => {
  const raw = String(value || "").trim();
  if (raw && /[^0-9+\-\s]/.test(raw)) {
    return {
      valid: false,
      normalized: "",
      message: "กรุณากรอกเฉพาะตัวเลข เว้นวรรคหรือเครื่องหมาย - เท่านั้น",
    };
  }
  if (raw.includes("+") && !raw.replace(/[\s-]/g, "").startsWith("+66")) {
    return {
      valid: false,
      normalized: "",
      message: "กรุณากรอกเบอร์โทรหรือเลขบัตรประชาชนที่ถูกต้อง",
    };
  }

  const normalized = normalizePromptPay(value);
  if (!normalized) return { valid: true, normalized: "" };

  const valid =
    (normalized.length === 10 && /^0\d{9}$/.test(normalized)) ||
    (normalized.length === 13 && /^\d{13}$/.test(normalized));

  return {
    valid,
    normalized,
    message: "กรุณากรอกพร้อมเพย์เป็นเบอร์มือถือ 10 หลักหรือเลขบัตรประชาชน 13 หลัก",
  };
};

export const openBankApp = async (textToCopy, scheme, title) => {
  let copied = false;
  if (navigator.clipboard?.writeText) {
    try {
      await navigator.clipboard.writeText(textToCopy);
      copied = true;
    } catch {
      copied = false;
    }
  } else {
    const textarea = document.createElement("textarea");
    textarea.value = textToCopy;
    textarea.style.position = "fixed";
    textarea.style.opacity = "0";
    document.body.appendChild(textarea);
    textarea.select();
    copied = document.execCommand("copy");
    textarea.remove();
  }

  if (!copied) {
    dynaura.error({
      title: "คัดลอกไม่สำเร็จ",
      description: "กรุณาลองคัดลอกเลขบัญชีอีกครั้ง",
      duration: 4000,
    });
    return;
  }

  dynaura.success({
    title,
    description: scheme
      ? "ก๊อปเลขบัญชีแล้ว กำลังเปิดแอปธนาคาร"
      : "คัดลอกเลขบัญชีเรียบร้อยแล้ว",
    duration: 3000,
  });

  if (scheme) window.location.href = scheme;
};
