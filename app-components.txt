//  Helper จัดการวันที่
const getTodayISO = () => new Date().toLocaleDateString("en-CA");
const formatForDB = (isoDate) => {
  if (!isoDate || !isoDate.includes("-")) return isoDate;
  const [y, m, d] = isoDate.split("-");
  return `${d}/${m}/${y}`;
};
const parseForUI = (thDate) => {
  if (!thDate || !thDate.includes("/")) return getTodayISO();
  const [d, m, y] = thDate.split("/");
  return `${y}-${m.padStart(2, "0")}-${d.padStart(2, "0")}`;
};
const getStoredMembers = () => {
  try {
    const saved = JSON.parse(localStorage.getItem("tripMembers") || "null");
    if (Array.isArray(saved)) {
      const members = [
        ...new Set(saved.map((member) => String(member).trim())),
      ].filter(Boolean);
      if (members.length > 0) return members;
    }
  } catch {
    localStorage.removeItem("tripMembers");
  }
  return DEFAULT_MEMBERS;
};
const getOfflineQueue = () => {
  try {
    const queue = JSON.parse(localStorage.getItem("offlineQueue") || "[]");
    return Array.isArray(queue) ? queue : [];
  } catch {
    localStorage.removeItem("offlineQueue");
    return [];
  }
};

      amount: ((end - start) / rate) * price,

      odometer_end: end,

    if (!navigator.onLine) {

        return Swal.fire({
          icon: "warning",
          title: "แก้ไขขณะออฟไลน์ไม่ได้",

        icon: "info",

        text: "บันทึกลงเครื่องแล้ว",

    } else {

          .from("trip_expenses")
          .update(payload)
          .eq("id", editId);

            icon: "error",

            text: error.message,

        const { error } = await supabase
          .from("trip_expenses")
          .insert([payload]);

          return Swal.fire({
            icon: "error",
            title: "บันทึกไม่สำเร็จ",
            text: error.message,

        title: "ออฟไลน์",
        text: "ไม่สามารถลบรายการค้างซิงก์ได้ รอเน็ตกลับมาก่อนนะครับ",

        icon: "error",
        title: "ออฟไลน์",
        text: "เน็ตหลุดจ้า",

      QR_IMAGES[to] || "https://placehold.co/250x250?text=No+QR+Code";

        icon: "error",
        title: "บันทึกการโอนไม่สำเร็จ",

      icon: "success",
      title: "เคลียร์ยอดสำเร็จ!",
      text: `บันทึกการโอนเงินของ ${from} แล้ว ยอดหนี้เป็น 0`,
      timer: 2000,
      showConfirmButton: false,
    });

    (expense) => expense.category !== HISTORY_CATEGORY,

    (expense) => expense.category === HISTORY_CATEGORY,

    () =>
      regularExpenses.reduce((groups, expense) => {

        return groups;
      }, {}),
    [regularExpenses],
  );
  const totalExpense = useMemo(
    () =>
      regularExpenses.reduce((sum, expense) => sum + Number(expense.amount), 0),
    [regularExpenses],
  );
  const perPerson = MEMBERS.length > 0 ? totalExpense / MEMBERS.length : 0;
  const previewGasCost =
    Number.isFinite(Number(odoStart)) &&
    Number.isFinite(Number(odoEnd)) &&
    Number.isFinite(Number(gasPrice)) &&
    Number(gasRate) > 0
      ? ((Number(odoEnd) - Number(odoStart)) / Number(gasRate)) *
        Number(gasPrice)
      : 0;
  const settlements = useMemo(() => {
    const balances = MEMBERS.map((member) => {
      const groupPaidAmount = regularExpenses
        .filter((expense) => expense.paid_by === member)
        .reduce((sum, expense) => sum + Number(expense.amount), 0);
      let balance = groupPaidAmount - perPerson;
      historyExpenses.forEach((expense) => {
        if (expense.paid_by === member) balance += Number(expense.amount);
        if (expense.item_name.includes(`-> ${member}`))
          balance -= Number(expense.amount);
      });
      return { member, balance, groupPaidAmount };
    });
    const debtors = balances
      .filter((balance) => balance.balance < -0.01)
      .map((balance) => ({ ...balance }));
    const creditors = balances
      .filter((balance) => balance.balance > 0.01)
      .map((balance) => ({ ...balance }));
    const transfers = [];
    let debtorIndex = 0;
    let creditorIndex = 0;
    while (debtorIndex < debtors.length && creditorIndex < creditors.length) {
      const debtor = debtors[debtorIndex];
      const creditor = creditors[creditorIndex];
      const transferAmount = Math.min(
        Math.abs(debtor.balance),
        creditor.balance,
      );
      if (transferAmount > 0.01)
        transfers.push({
          from: debtor.member,
          to: creditor.member,
          amount: transferAmount,
        });
      debtor.balance += transferAmount;
      creditor.balance -= transferAmount;
      if (Math.abs(debtor.balance) < 0.01) debtorIndex++;
      if (creditor.balance < 0.01) creditorIndex++;
    }
    return { balances, transfers };
  }, [MEMBERS, regularExpenses, historyExpenses, perPerson]);

  return (
    <div className="relative min-h-screen bg-black pb-20 font-sans text-gray-100">
      <TripHeader
        totalExpense={totalExpense}
        perPerson={perPerson}
        isSyncing={isSyncing}
        onRefresh={refreshExpenses}

          editId={editId}
          date={date}
          category={category}
          categories={CATEGORIES}
          item={item}
          amount={amount}
          paidBy={paidBy}
          members={MEMBERS}
          onDateChange={setDate}
          onCategoryChange={setCategory}
          onItemChange={setItem}
          onAmountChange={setAmount}
          onPaidByChange={handlePaidByChange}
          onAddMember={handleAddMember}
          onSubmit={handleAddExpense}
          onCancel={cancelEdit}

          expenses={regularExpenses}
          groupedExpenses={groupedExpenses}
          loading={loading}
          isExpanded={isListExpanded}
          collapsedDates={collapsedDates}
          onToggleExpanded={() => setIsListExpanded((expanded) => !expanded)}
          onToggleDate={(value) =>
            setCollapsedDates((dates) => ({ ...dates, [value]: !dates[value] }))
          }
          onEdit={handleEdit}
          onDelete={handleDelete}
        />
        <SettlementCard
          settlements={settlements}
          handleRemoveMember={handleRemoveMember}
          onSettle={handleSettleDebt}
        />
        <GasForm
          odoStart={odoStart}
          odoEnd={odoEnd}
          gasPrice={gasPrice}
          gasRate={gasRate}
          previewGasCost={previewGasCost}
          onOdoStartChange={setOdoStart}
          onOdoEndChange={setOdoEnd}
          onGasPriceChange={setGasPrice}
          onGasRateChange={setGasRate}
          onSubmit={handleAddGas}
        />
      </main>
      <button
        onClick={() => setShowHistory(true)}
        className="fixed bottom-6 right-6 z-40 rounded-full bg-amber-500 p-4 text-gray-950 shadow-xl shadow-lime-500/10 transition hover:bg-amber-700"
        title="ดูประวัติการเคลียร์เงิน"

        <HistoryModal

          onClose={() => setShowHistory(false)}
          onUndo={handleUndoHistory}
        />
      )}
    </div>
  );
}

import {
  ChevronDown,
  ChevronRight,
  Edit2,
  FolderClosed,
  FolderOpen,
  Maximize2,
  Minimize2,
  Receipt,
  Trash2,
} from "lucide-react";
import { useState, useRef } from "react";

//  Component ย่อยสำหรับแต่ละรายการ
function ExpenseItem({
  expense,
  swipedId,
  setSwipedId,
  onEdit,
  onDelete,
  formatMoney,
  isLast,
}) {
  const [drag, setDrag] = useState({
    active: false,
    startX: 0,
    startY: 0,
    offset: 0,
    scrolling: false,
  });
  const isMouse = useRef(false);

  const isOpen = swipedId === expense.id;

  // ปรับระยะดึงให้พอดีกับปุ่มวงกลม (ปุ่มวงกลมใช้พื้นที่ประมาณ 60px)
  let currentOffset = isOpen ? -70 : 0;
  if (drag.active) {
    currentOffset = (isOpen ? -70 : 0) + drag.offset;
    if (currentOffset > 0) currentOffset = 0;
    if (currentOffset < -90) currentOffset = -90; // ยืดสุดได้แค่ -90px
  }

  // เช็คว่าเริ่มโดนลากเกิน 5px หรือยัง เพื่อเปลี่ยนขอบการ์ดให้โค้งมน
  const isSwiped = currentOffset < -5;

  const startDrag = (clientX, clientY) => {
    setDrag({
      active: true,
      startX: clientX,
      startY: clientY,
      offset: 0,
      scrolling: false,
    });
    if (swipedId !== expense.id) setSwipedId(null);
  };

  const moveDrag = (clientX, clientY) => {
    if (!drag.active) return;
    const diffX = clientX - drag.startX;
    const diffY = Math.abs(clientY - drag.startY);

    if (!drag.scrolling && diffY > 15 && Math.abs(diffX) < 15) {
      setDrag((prev) => ({ ...prev, scrolling: true }));
    }
    if (drag.scrolling) return;
    setDrag((prev) => ({ ...prev, offset: diffX }));
  };

  const endDrag = () => {
    if (!drag.active) return;
    if (!drag.scrolling) {
      // ถ้าดึงเกิน -35 ให้ถือว่าเปิด
      if (currentOffset < -35) setSwipedId(expense.id);
      else setSwipedId(null);
    }
    setDrag({
      active: false,
      startX: 0,
      startY: 0,
      offset: 0,
      scrolling: false,
    });
  };

  return (
    // เปลี่ยนพื้นหลังหลักเป็นสีเดียวกับฉากหลัง เพื่อให้ปุ่มวงกลมดูลอยเด่นขึ้นมา
    <div
      className={`relative bg-gray-900 ${!isLast ? "border-b border-gray-800" : ""}`}
    >
      {/* 🗑️ กล่องปุ่ม Delete "วงกลม" สีแดง ลอยอยู่ขวาสุด */}
      <div className="absolute inset-y-0 right-3 flex items-center justify-center">
        <button
          onClick={() => {
            onDelete(expense.id);
            setSwipedId(null);
          }}
          //  กำหนดให้ปุ่มเป็นวงกลม (h-11 w-11 rounded-full)
          className="flex h-13 w-13 items-center justify-center rounded-full bg-red-500 text-white shadow-sm transition hover:bg-red-600"
          style={{
            // ถ้ายังดึงไม่ถึงระยะ ให้ซ่อนปุ่มไว้ก่อน (opacity) และให้มันค่อยๆ ขยายตัวโผล่มา (scale)
            opacity: currentOffset < -20 ? 1 : 0,
            transform: `scale(${Math.min(1.2, 0.2 + (-currentOffset / 70) * 0.8)})`,
            transition: drag.active ? "none" : "all 0.25s ease-out",
          }}
        >
          <Trash2 size={22} />
        </button>
      </div>

      {/* 📱 เลเยอร์การ์ดสีเทาหลัก */}
      <div
        className={` relative z-10 flex cursor-grab touch-pan-y items-center justify-between  p-3 pl-8 active:cursor-grabbing ${isOpen || drag.active ? "bg-gray-700" : "bg-gray-800"}  `}
        style={{
          transform: `translate3d(${currentOffset}px, 0, 0)`,
          transition: drag.active ? "none" : "all 0.25s ease-out",
          //  สั่งให้ขอบขวา "โค้งมน (Rounded)" เมื่อเริ่มโดนลาก
          borderTopRightRadius: isSwiped ? "20px" : "0px",
          borderBottomRightRadius: isSwiped ? "20px" : "0px",
        }}
        // Touch
        onTouchStart={(e) =>
          startDrag(e.targetTouches[0].clientX, e.targetTouches[0].clientY)
        }
        onTouchMove={(e) =>
          moveDrag(e.targetTouches[0].clientX, e.targetTouches[0].clientY)
        }
        onTouchEnd={endDrag}
        // Mouse
        onMouseDown={(e) => {
          isMouse.current = true;
          startDrag(e.clientX, e.clientY);
        }}
        onMouseMove={(e) => {
          if (isMouse.current) moveDrag(e.clientX, e.clientY);
        }}
        onMouseUp={() => {
          if (isMouse.current) {
            isMouse.current = false;
            endDrag();
          }
        }}
        onMouseLeave={() => {
          if (isMouse.current) {
            isMouse.current = false;
            endDrag();
          }
        }}
      >
        <div className="flex-1 pointer-events-none">
          <p className="mb-0.5 text-xs text-gray-400">
            จ่ายโดย:{" "}
            <span className="font-bold text-lime-400">{expense.paid_by}</span>
          </p>
          <p className="font-medium text-gray-100">{expense.item_name}</p>
          <p className="text-xs text-gray-400">{expense.category}</p>
        </div>
        <div className="flex items-center gap-3">
          <p className="font-bold text-gray-100 pointer-events-none">
            ฿{formatMoney(expense.amount)}
          </p>
          <button
            onClick={(e) => {
              e.stopPropagation();
              onEdit(expense);
              setSwipedId(null);
            }}
            className="text-gray-400 transition hover:text-lime-400 z-20 relative"
          >
            <Edit2 size={16} />
          </button>
        </div>
      </div>
    </div>
  );
}

          </p>
        ) : (
          settlements.transfers.map((transfer) => (
            <div
              key={`${transfer.from}-${transfer.to}`}
              className="rounded-lg border border-lime-900 bg-gray-800 p-3 "
            >
              <div className="mb-2 flex items-center justify-between">
                <div className="flex items-center gap-2 font-medium p-2 ">
                  <span className="text-gray-100">{transfer.from}</span>
                  <ArrowRight size={16} className="text-gray-400" />
                  <span className="text-gray-100">{transfer.to}</span>
                </div>
                <div className="font-bold text-lime-400">
                  ฿{formatMoney(transfer.amount)}
                </div>
              </div>
              <button
                onClick={() =>
                  onSettle(transfer.from, transfer.to, transfer.amount)
                }
                className="flex w-10/12 m-auto items-center justify-center gap-1 rounded bg-lime-500 py-2 text-xs text-gray-950 shadow-sm transition hover:bg-lime-300"
              >
                <CheckCircle size={14} /> เคลียร์ยอดนี้
              </button>
            </div>
          ))
        )}
      </div>
      <hr className="my-4 border-gray-800" />
      <div className="space-y-3">
        <p className="mb-2 text-xs font-bold text-gray-400">
          สรุปยอดสุทธิแต่ละคน
        </p>
        {settlements.balances.map((balance) => (
          <div
            key={balance.member}
            className="flex items-center justify-between text-sm"
          >
            <div className="flex items-center gap-2 font-medium text-gray-100">
              <span>{balance.member}</span>
              <button
                type="button"
                onClick={() => handleRemoveMember(balance.member)}
                className="rounded p-1 text-red-500 transition hover:bg-red-100 hover:text-red-700"
                title={`ลบ ${balance.member}`}
                aria-label={`ลบ ${balance.member}`}
              >
                <Trash2 size={14} />
              </button>
            </div>
            <div className="text-right">
              {balance.balance > 0.01 ? (
                <span className="font-bold text-lime-400">
                  (ได้คืน ฿{formatMoney(balance.balance)})
                </span>
              ) : balance.balance < -0.01 ? (
                <span className="font-bold text-red-500">
                  (ติดลบ ฿{formatMoney(Math.abs(balance.balance))})
                </span>
              ) : (
                <span className="font-bold text-gray-400">(พอดี)</span>
              )}
            </div>
          </div>
        ))}
      </div>
    </section>
  );
}

// src/components/TripHeader.jsx

import { RefreshCw, Check } from "lucide-react";
import { useEffect, useState } from "react";

export default function TripHeader({ totalExpense, perPerson, isSyncing, onRefresh, MEMBERS }) {
  const [syncStatus, setSyncStatus] = useState("idle");

  // 1. ดักจับการเริ่มซิงก์และซิงก์เสร็จ
  useEffect(() => {
    if (isSyncing) {
      setSyncStatus("syncing");
    } else if (!isSyncing && syncStatus === "syncing") {
      setSyncStatus("success");
    }
  }, [isSyncing]); // เอา syncStatus ออกจาก dep จะได้ไม่เผลอไปตัด Timer ตัวเอง

  // 2. ดักจับตอนเป็น success ให้รอ 1.5 วิ แล้วค่อยกลับไป idle
  useEffect(() => {
    if (syncStatus === "success") {
      const timer = setTimeout(() => setSyncStatus("idle"), 1500); // เพิ่มเวลาให้ดูแอนิเมชันทัน
      return () => clearTimeout(timer);
    }
  }, [syncStatus]);

  const formatMoney = (value) =>
    value.toLocaleString(undefined, {
      minimumFractionDigits: 2,
      maximumFractionDigits: 2,
    });

  return (
    <header className="sticky top-0 z-40 bg-gray-900 p-4 text-gray-100 shadow-md">
      {/*  แทรก CSS สำหรับทำแอนิเมชันวาดเส้นขีดถูก */}
      <style>{`
        @keyframes drawCheck {
          0% {
            stroke-dasharray: 50;
            stroke-dashoffset: 50;
          }
          100% {
            stroke-dasharray: 50;
            stroke-dashoffset: 0;
          }
        }
        .animate-draw {
          animation: drawCheck 0.4s ease-out ;
        }
      `}</style>

      <div className="relative mx-auto max-w-md text-center">
        <button
          onClick={onRefresh}
          disabled={syncStatus === "syncing"} 
          // 🧠 ล็อกความกว้าง-สูงให้ปุ่ม (h-10 w-10) ป้องกันปุ่มขยับตอนสลับไอคอน
          className="absolute right-5 top-5 flex h-10 w-10 items-center justify-center rounded-full bg-gray-800 transition hover:bg-gray-700"
          title="รีเฟรชข้อมูล"
        >
          {syncStatus === "syncing" ? (
            <RefreshCw size={40} className="animate-spin text-gray-400" />
          ) : syncStatus === "success" ? (
            //  เรียกใช้ class animate-draw ที่เราสร้างไว้ใน <style>
            <Check size={40} className="text-lime-400 animate-draw" />
          ) : (
            <RefreshCw size={40} className="text-gray-300 transition" />
          )}
        </button>
        <p className="mt-1 text-sm font-light opacity-80">ยอดรวมค่าใช้จ่ายทั้งทริป</p>
        <h1 className="my-1 text-4xl font-bold">฿{formatMoney(totalExpense)}</h1>
        <p className="mt-3 inline-block rounded-full bg-gray-800 px-3 py-1 text-sm text-lime-400">
          หาร {MEMBERS.length} คน = ตกคนละ ฿{formatMoney(perPerson)}
        </p>
      </div>
    </header>
  );
}
